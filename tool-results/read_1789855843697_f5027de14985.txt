    13→    13→function rango(f: Filtros): { gte?: Date; lte?: Date } {
    14→    14→  const r: { gte?: Date; lte?: Date } = {}
    15→    15→  if (f.from) r.gte = new Date(f.from + 'T00:00:00.000Z')
    16→    16→  if (f.to) r.lte = new Date(f.to + 'T23:59:59.999Z')
    17→    17→  return r
    18→    18→}

    21→    21→// ============ RESUMEN EJECUTIVO ============
    22→    22→export async function getResumen() {
    23→    23→  const [olaCount, h61Count, tmCount, pickCount, batches] = await Promise.all([
    24→    24→    db.olaDia.aggregate({ _count: true, _min: { fecha: true }, _max: { fecha: true }, _avg: { ola: true, total: true } }),
    25→    25→    db.h61OpDia.aggregate({ _count: true, _min: { fecha: true }, _max: { fecha: true }, _sum: { bultos: true, horasActivas: true, extras: true, bultosExtras: true } }),
    26→    26→    db.tiempoMuerto.aggregate({ _count: true, _min: { fecha: true }, _max: { fecha: true }, _sum: { minutos: true, minutosEf: true } }),
    27→    27→    db.pickingEvento.aggregate({ _count: true, _min: { fecha: true }, _max: { fecha: true } }),
    28→    28→    db.uploadBatch.findMany({ orderBy: { createdAt: 'desc' }, take: 10 }),
    29→    29→  ])
    30→    30→
    31→    31→  const opsUnicos = await db.h61OpDia.findMany({ select: { operario: true }, distinct: ['operario'] })
    32→    32→
    33→    33→  // top categorias TM (excluye bajas ESTADO B; OR con null es clave: NOT excludes NULLs en SQL)
    34→    34→  const tmCats = await db.tiempoMuerto.groupBy({
    35→    35→    by: ['categoria'],
    36→    36→    _sum: { minutosEf: true },
    37→    37→    _count: true,
    38→    38→    where: { OR: [{ estado: null }, { estado: { not: 'B' } }] },
    39→    39→    orderBy: { _sum: { minutosEf: 'desc' } },
    40→    40→  })
    41→    41→
    42→    42→  const horasTotales = h61Count._sum.horasActivas ?? 0
    43→    43→  const bultos = h61Count._sum.bultos ?? 0
    44→    44→  const bultosExtras = h61Count._sum.bultosExtras ?? 0
    45→    45→  const totalMinTM = tmCats.reduce((a, c) => a + (c._sum.minutosEf ?? 0), 0)
    46→    46→
    47→    47→  return {
    48→    48→    datasets: {
    49→    49→      ola: { registros: olaCount._count, desde: olaCount._min.fecha, hasta: olaCount._max.fecha, olaMedia: Math.round(olaCount._avg.ola ?? 0), totalMedia: Math.round(olaCount._avg.total ?? 0) },
    50→    50→      h61: { registros: h61Count._count, desde: h61Count._min.fecha, hasta: h61Count._max.fecha, bultos, horas: horasTotales, productividad: horasTotales ? +(bultos / horasTotales).toFixed(1) : 0, operarios: opsUnicos.length, extrasHoras: h61Count._sum.extras ?? 0, bultosExtras, pctBultosExtras: bultos ? +((bultosExtras / bultos) * 100).toFixed(1) : 0 },
    51→    51→      tm: { registros: tmCount._count, desde: tmCount._min.fecha, hasta: tmCount._max.fecha, minutos: totalMinTM, horas: +(totalMinTM / 60).toFixed(0), pctSobreHorasH61: horasTotales ? +(((totalMinTM / 60) / horasTotales) * 100).toFixed(1) : 0 },
    52→    52→      picking: { registros: pickCount._count, desde: pickCount._min.fecha, hasta: pickCount._max.fecha },
    53→    53→    },
    54→    54→    topCategoriasTM: tmCats.slice(0, 6).map((c) => ({ categoria: c.categoria, minutos: c._sum.minutosEf ?? 0, registros: c._count })),
    55→    55→    batches: batches.map((b) => ({ ...b, createdAt: b.createdAt.toISOString() })),
    56→    56→  }
    57→    57→}

    60→    60→export async function getPlanificacion(f: Filtros) {
    61→    61→  const where = { fecha: rango(f) }
    62→    62→  const [ola, h61] = await Promise.all([
    63→    63→    db.olaDia.findMany({ where, orderBy: { fecha: 'asc' } }),
    64→    64→    db.h61OpDia.findMany({ where: { fecha: rango(f) }, select: { fecha: true, operario: true, horasActivas: true, bultos: true, bultosBase: true, bultosExtras: true, extras: true, turno: true, funcion: true } }),
    65→    65→  ])
    66→    66→
    67→    67→  const porDia = new Map<string, { ops: number; opsExtras: number; horas: number; horasExtras: number; bultos: number; bultosBase: number; bultosExtras: number }>()
    68→    68→  for (const r of h61) {
    69→    69→    const k = dia(r.fecha)
    70→    70→    let p = porDia.get(k)
    71→    71→    if (!p) { p = { ops: 0, opsExtras: 0, horas: 0, horasExtras: 0, bultos: 0, bultosBase: 0, bultosExtras: 0 }; porDia.set(k, p) }
    72→    72→    p.ops += 1
    73→    73→    if (r.extras > 0) p.opsExtras += 1
    74→    74→    p.horas += r.horasActivas
    75→    75→    p.horasExtras += r.extras
    76→    76→    p.bultos += r.bultos
    77→    77→    p.bultosBase += r.bultosBase
    78→    78→    p.bultosExtras += r.bultosExtras
    79→    79→  }
    80→    80→
    81→    81→  const fechas = new Set<string>([...ola.map((o) => dia(o.fecha)), ...porDia.keys()])
    82→    82→  const serie = [...fechas].sort().map((k) => {
    83→    83→    const o = ola.find((x) => dia(x.fecha) === k)
    84→    84→    const p = porDia.get(k)
    85→    85→    const demanda = o?.total ?? null
    86→    86→    const prodHora = p && p.horas ? p.bultos / p.horas : null
    87→    87→    // capacidad si se suprimen extras: mantiene bultosBase (produccion en <=8h por operario)
    88→    88→    const faltanteSinExtras = p && demanda != null ? Math.max(0, demanda - p.bultosBase) : null
    89→    89→    return {
    90→    90→      fecha: k,
    91→    91→      ola: o?.ola ?? null,
    92→    92→      pendiente: o?.pendiente ?? null,
    93→    93→      demanda,
    94→    94→      preparado: p?.bultos ?? null,
    95→    95→      ops: p?.ops ?? null,
    96→    96→      opsExtras: p?.opsExtras ?? null,
    97→    97→      horas: p ? +p.horas.toFixed(1) : null,
    98→    98→      horasExtras: p?.horasExtras ?? 0,
    99→    99→      prodHora: prodHora != null ? +prodHora.toFixed(1) : null,
   100→   100→      bultosBase: p?.bultosBase ?? null,
   101→   101→      bultosExtras: p?.bultosExtras ?? null,
   102→   102→      faltanteSinExtras: faltanteSinExtras != null ? Math.round(faltanteSinExtras) : null,
   103→   103→    }
   104→   104→  })
   105→   105→
   106→   106→  // agregados por mes
   107→   107→  const porMes = new Map<string, { ola: number; demanda: number; preparado: number; dias: number; diasExtras: number; horasExtras: number; faltante: number }>()
   108→   108→  for (const s of serie) {
   109→   109→    const k = s.fecha.slice(0, 7)
   110→   110→    let m = porMes.get(k)
   111→   111→    if (!m) { m = { ola: 0, demanda: 0, preparado: 0, dias: 0, diasExtras: 0, horasExtras: 0, faltante: 0 }; porMes.set(k, m) }
   112→   112→    m.dias += 1
   113→   113→    if (s.demanda != null) m.demanda += s.demanda
   114→   114→    if (s.ola != null) m.ola += s.ola
   115→   115→    if (s.preparado != null) m.preparado += s.preparado
   116→   116→    if ((s.opsExtras ?? 0) > 0) m.diasExtras += 1
   117→   117→    m.horasExtras += s.horasExtras ?? 0
   118→   118→    if (s.faltanteSinExtras != null) m.faltante += s.faltanteSinExtras
   119→   119→  }
   120→   120→  const meses = [...porMes.entries()].sort().map(([mes, v]) => ({ mes, ...v }))
   121→   121→
   122→   122→  const conExtras = serie.filter((s) => (s.opsExtras ?? 0) > 0)
   123→   123→  const resumen = {
   124→   124→    dias: serie.length,
   125→   125→    diasConExtras: conExtras.length,
   126→   126→    pctDiasConExtras: serie.length ? +((conExtras.length / serie.length) * 100).toFixed(1) : 0,
   127→   127→    faltantePromedioSinExtras: conExtras.length ? Math.round(conExtras.reduce((a, s) => a + (s.faltanteSinExtras ?? 0), 0) / conExtras.length) : 0,
   128→   128→    faltanteTotalSinExtras: conExtras.reduce((a, s) => a + (s.faltanteSinExtras ?? 0), 0),
   129→   129→    horasExtrasTotales: serie.reduce((a, s) => a + (s.horasExtras ?? 0), 0),
   130→   130→    bultosExtrasTotales: serie.reduce((a, s) => a + (s.bultosExtras ?? 0), 0),
   131→   131→    prodHoraMedia: (() => {
   132→   132→      const vals = serie.filter((s) => s.prodHora != null && s.preparado != null && (s.preparado ?? 0) > 0)
   133→   133→      const totB = vals.reduce((a, s) => a + (s.preparado ?? 0), 0)
   134→   134→      const totH = vals.reduce((a, s) => a + (s.horas ?? 0), 0)
   135→   135→      return totH ? +(totB / totH).toFixed(1) : null
   136→   136→    })(),
   137→   137→  }
   138→   138→
   139→   139→  return { serie, meses, resumen }
   140→   140→}

   142→   142→// ============ OLA (analisis exclusivo de ola y pendientes) ============
   143→   143→// Serie diaria cruda + lunes de cada semana ISO. Los agregados (mes/semana/dia de
   144→   144→// semana, promedio y mediana) se recalculan en el cliente para que los filtros
   145→   145→// mensual/semanal/diario sean consistentes entre si.
   146→   146→const DIAS_SEM = ['Domingo', 'Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado']
   147→   147→
   148→   148→export async function getOla(f: Filtros) {
   149→   149→  const rows = await db.olaDia.findMany({ where: { fecha: rango(f) }, orderBy: { fecha: 'asc' } })
   150→   150→  const serie = rows.map((r) => {
   151→   151→    const k = dia(r.fecha)
   152→   152→    const d = new Date(k + 'T00:00:00.000Z')
   153→   153→    const dow = d.getUTCDay() // 0=domingo
   154→   154→    const lun = new Date(d)
   155→   155→    lun.setUTCDate(lun.getUTCDate() - ((dow + 6) % 7)) // lunes de la semana
   156→   156→    const ola = r.ola ?? 0
   157→   157→    const pendiente = r.pendiente ?? 0
   158→   158→    return {
   159→   159→      fecha: k,
   160→   160→      diaSemana: DIAS_SEM[dow],
   161→   161→      esFinde: dow === 0 || dow === 6,
   162→   162→      semana: lun.toISOString().slice(0, 10),
   163→   163→      ola,
   164→   164→      pendiente,
   165→   165→      total: ola + pendiente,
   166→   166→    }
   167→   167→  })
   168→   168→  return { serie }
   169→   169→}

   171→   171→// ============ PRODUCTIVIDAD H61 ============
   172→   172→export async function getH61(f: Filtros) {
   173→   173→  const where: Record<string, unknown> = { fecha: rango(f) }
   174→   174→  if (f.turno) where.turno = f.turno
   175→   175→  if (f.funcion) where.funcion = f.funcion
   176→   176→
   177→   177→  const [ops, th, ci] = await Promise.all([
   178→   178→    db.h61OpDia.findMany({ where }),
   179→   179→    db.h61TurnoHora.findMany({ where: { fecha: rango(f) } }),
   180→   180→    db.h61Circuito.findMany({ where: { fecha: rango(f) } }),
   181→   181→  ])
   182→   182→
   183→   183→  // Perfil 24h: horas 0-5 del turno noche (N) pertenecen al dia calendario siguiente
   184→   184→  const porHora = Array.from({ length: 24 }, () => ({ bultos: 0, operarios: new Set<string>() }))
   185→   185→  const opSeen = new Map<string, Set<number>>() // fecha|op -> horas
   186→   186→  for (const r of th) {
   187→   187→    const fechaBase = new Date(r.fecha)
   188→   188→    const esNocheMadrugada = r.turno === 'N' && r.hora <= 5
   189→   189→    if (esNocheMadrugada) fechaBase.setUTCDate(fechaBase.getUTCDate() + 1)
   190→   190→    const h = porHora[r.hora]
   191→   191→    h.bultos += r.bultos
   192→   192→    const k = `${dia(fechaBase)}|${r.turno}`
   193→   193→    // operarios por hora aproximamos con el maximo informado ese dia-turno
   194→   194→    const set = opSeen.get(k) ?? new Set<number>()
   195→   195→    set.add(r.operarios)
   196→   196→    opSeen.set(k, set)
   197→   197→  }
   198→   198→  const perfilHora = porHora.map((h, i) => ({
   199→   199→    hora: i,
   200→   200→    etiqueta: `${String(i).padStart(2, '0')}:00`,
   201→   201→    bultos: h.bultos,
   202→   202→    operariosProm: (() => {
   203→   203→      const vals = [...opSeen.values()].map((s) => Math.max(...s))
   204→   204→      return vals.length ? +(vals.reduce((a, b) => a + b, 0) / vals.length).toFixed(1) : 0
   205→   205→    })(),
   206→   206→  }))
   207→   207→  const horasConDatos = perfilHora.filter((p) => p.bultos > 0)
   208→   208→  const pico = horasConDatos.reduce((a, b) => (b.bultos > a.bultos ? b : a), horasConDatos[0] ?? { hora: 0, etiqueta: '-', bultos: 0, operariosProm: 0 })
   209→   209→  const valle = horasConDatos.reduce((a, b) => (b.bultos < a.bultos ? b : a), horasConDatos[0] ?? { hora: 0, etiqueta: '-', bultos: 0, operariosProm: 0 })
   210→   210→
   211→   211→  // por turno
   212→   212→  const turnoAgg = new Map<string, { bultos: number; horas: number; opsDias: number; bultosExtras: number }>()
   213→   213→  for (const r of th) {
   214→   214→    let t = turnoAgg.get(r.turno)
   215→   215→    if (!t) { t = { bultos: 0, horas: 0, opsDias: 0, bultosExtras: 0 }; turnoAgg.set(r.turno, t) }
   216→   216→    t.bultos += r.bultos
   217→   217→  }
   218→   218→  const turnoHoras = new Map<string, Set<string>>()
   219→   219→  for (const r of ops) {
   220→   220→    let t = turnoHoras.get(r.turno)
   221→   221→    if (!t) { t = new Set(); turnoHoras.set(r.turno, t) }
   222→   222→    t.add(`${dia(r.fecha)}|${r.operario}`)
   223→   223→    t.add(`${dia(r.fecha)}|${r.operario}|h|${r.horasActivas}`)
   224→   224→  }
   225→   225→  for (const [turno, set] of turnoHoras) {
   226→   226→    const horas = [...set].filter((x) => x.includes('|h|')).reduce((a, x) => a + parseInt(x.split('|h|')[1], 10), 0)
   227→   227→    const opsDias = [...set].filter((x) => !x.includes('|h|')).length
   228→   228→    const t = turnoAgg.get(turno)
   229→   229→    if (t) { t.horas = horas; t.opsDias = opsDias }
   230→   230→  }
   231→   231→  for (const r of ops) {
   232→   232→    const t = turnoAgg.get(r.turno)
   233→   233→    if (t) t.bultosExtras += r.bultosExtras
   234→   234→  }
   235→   235→  const porTurno = [...turnoAgg.entries()].map(([turno, v]) => ({
   236→   236→    turno,
   237→   237→    nombre: turno === 'M' ? 'Mañana' : turno === 'T' ? 'Tarde' : turno === 'N' ? 'Noche' : turno,
   238→   238→    ...v,
   239→   239→    prodHora: v.horas ? +(v.bultos / v.horas).toFixed(1) : 0,
   240→   240→  })).sort((a, b) => b.bultos - a.bultos)
   241→   241→
   242→   242→  // heatmap turno x hora
   243→   243→  const heat = ['M', 'T', 'N'].map((turno) => {
   244→   244→    const fila = Array.from({ length: 24 }, () => 0)
   245→   245→    for (const r of th) if (r.turno === turno) fila[r.hora] = Math.max(fila[r.hora], Math.round(r.bultos / Math.max(1, r.operarios)))
   246→   246→    return { turno, valores: fila }
   247→   247→  })
   248→   248→
   249→   249→  // circuitos
   250→   250→  const ciAgg = new Map<string, number>()
   251→   251→  for (const r of ci) ciAgg.set(r.circuito, (ciAgg.get(r.circuito) ?? 0) + r.bultos)
   252→   252→  const circuitos = [...ciAgg.entries()].map(([circuito, bultos]) => ({ circuito, bultos })).sort((a, b) => b.bultos - a.bultos)
   253→   253→
   254→   254→  // distribucion de jornadas 8h vs extras
   255→   255→  const distHoras = new Map<number, number>()
   256→   256→  for (const r of ops) distHoras.set(r.horasActivas, (distHoras.get(r.horasActivas) ?? 0) + 1)
   257→   257→  const distribucion = [...distHoras.entries()].sort((a, b) => a[0] - b[0]).map(([horas, opsDia]) => ({ horas, opsDia }))
   258→   258→
   259→   259→  const opsExtras = ops.filter((r) => r.extras > 0)
   260→   260→  const extrasResumen = {
   261→   261→    opsDiasTotal: ops.length,
   262→   262→    opsDiasConExtras: opsExtras.length,
   263→   263→    pctOpsDiasConExtras: ops.length ? +((opsExtras.length / ops.length) * 100).toFixed(1) : 0,
   264→   264→    horasExtras: opsExtras.reduce((a, r) => a + r.extras, 0),
   265→   265→    bultosExtras: opsExtras.reduce((a, r) => a + r.bultosExtras, 0),
   266→   266→    bultosTotales: ops.reduce((a, r) => a + r.bultos, 0),
   267→   267→  }
   268→   268→  extrasResumen.pctBultosExtras = extrasResumen.bultosTotales ? +((extrasResumen.bultosExtras / extrasResumen.bultosTotales) * 100).toFixed(1) : 0
   269→   269→
   270→   270→  // top operarios por productividad (minimo 80 horas)
   271→   271→  const opAgg = new Map<string, { nombre: string | null; bultos: number; horas: number; extras: number; dias: number; funcion: string }>()
   272→   272→  for (const r of ops) {
   273→   273→    let o = opAgg.get(r.operario)
   274→   274→    if (!o) { o = { nombre: r.nombre, bultos: 0, horas: 0, extras: 0, dias: 0, funcion: r.funcion }; opAgg.set(r.operario, o) }
   275→   275→    if (!o.nombre && r.nombre) o.nombre = r.nombre
   276→   276→    o.bultos += r.bultos
   277→   277→    o.horas += r.horasActivas
   278→   278→    o.extras += r.extras
   279→   279→    o.dias += 1
   280→   280→  }
   281→   281→  const operarios = [...opAgg.entries()]
   282→   282→    .map(([operario, o]) => ({ operario, nombre: o.nombre ?? operario, bultos: o.bultos, horas: o.horas, extras: o.extras, dias: o.dias, prodHora: o.horas ? +(o.bultos / o.horas).toFixed(1) : 0 }))
   283→   283→    .filter((o) => o.horas >= 80)
   284→   284→    .sort((a, b) => b.prodHora - a.prodHora)
   285→   285→
   286→   286→  // serie diaria
   287→   287→  const serDia = new Map<string, { bultos: number; horas: number; ops: number }>()
   288→   288→  for (const r of ops) {
   289→   289→    const k = dia(r.fecha)
   290→   290→    let s = serDia.get(k)
   291→   291→    if (!s) { s = { bultos: 0, horas: 0, ops: 0 }; serDia.set(k, s) }
   292→   292→    s.bultos += r.bultos
   293→   293→    s.horas += r.horasActivas
   294→   294→    s.ops += 1
   295→   295→  }
   296→   296→  const serie = [...serDia.entries()].sort().map(([fecha, v]) => ({ fecha, ...v, prodHora: v.horas ? +(v.bultos / v.horas).toFixed(1) : 0 }))
   297→   297→
   298→   298→  const bultosTot = ops.reduce((a, r) => a + r.bultos, 0)
   299→   299→  const horasTot = ops.reduce((a, r) => a + r.horasActivas, 0)
   300→   300→
   301→   301→  return {
   302→   302→    perfilHora,
   303→   303→    pico: { ...pico },
   304→   304→    valle: { ...valle },
   305→   305→    porTurno,
   306→   306→    heat,
   307→   307→    circuitos: circuitos.slice(0, 17),
   308→   308→    distribucion,
   309→   309→    extrasResumen,
   310→   310→    operariosTop: operarios.slice(0, 15),
   311→   311→    operariosBottom: operarios.slice(-15).reverse(),
   312→   312→    operariosExtras: [...opAgg.entries()].map(([operario, o]) => ({ operario, nombre: o.nombre ?? operario, extras: o.extras, dias: o.dias })).filter((o) => o.extras > 0).sort((a, b) => b.extras - a.extras).slice(0, 15),
   313→   313→    serie,
   314→   314→    resumen: { bultos: bultosTot, horas: horasTot, prodHora: horasTot ? +(bultosTot / horasTot).toFixed(1) : 0, opsUnicos: opAgg.size },
   315→   315→  }
   316→   316→}
   317→   317→
   318→   318→// ============ CAPACIDAD H61 (ritmo de preparacion: jornada vs extras) ============
   319→   319→// Regla: las primeras 8 horas con produccion de cada operario-dia cuentan como
   320→   320→// jornada base; el excedente de horas y sus bultos cuentan como horas extra.
   321→   321→function mediana(vals: number[]): number {
   322→   322→  if (!vals.length) return 0
   323→   323→  const s = [...vals].sort((a, b) => a - b)
   324→   324→  const mid = Math.floor(s.length / 2)
   325→   325→  return s.length % 2 ? s[mid] : (s[mid - 1] + s[mid]) / 2
   326→   326→}

   328→   328→export async function getCapacidad(f: Filtros) {
   329→   329→  const [ops, oh] = await Promise.all([
   330→   330→    db.h61OpDia.findMany({ where: { fecha: rango(f) }, orderBy: { fecha: 'asc' } }),
   331→   331→    db.h61OpHora.findMany({ where: { fecha: rango(f) } }),
   332→   332→  ])
   333→   333→  if (!ops.length) return { serie: [], porMes: [], porTurno: [], perfilHora: [], resumen: null, feriados: [], tieneOpHora: oh.length > 0 }
   334→   334→
   335→   335→  // --- feriados argentinos (se sincronizan on-demand desde api.argentinadatos.com.ar) ---
   336→   336→  const anios = new Set<number>()
   337→   337→  for (const r of ops) {
   338→   338→    const y = r.fecha.getUTCFullYear()
   339→   339→    anios.add(y)
   340→   340→    // un TN de diciembre puede pertenecer al feriado del 1° de enero siguiente
   341→   341→    if (r.turno === 'N' && r.fecha.getUTCMonth() === 11) anios.add(y + 1)
   342→   342→  }
   343→   343→  const feriadosMap = await getFeriadosMap([...anios])
   344→   344→
   345→   345→  // Regla de feriados: en una jornada feriada TODO el tiempo y sus bultos
   346→   346→  // cuentan como horas extra y se miden aparte (no influyen en promedios,
   347→   347→  // medianas, perfiles ni tabla por turno de la medición normal).
   348→   348→  // Turnos diurnos (M/T): la jornada es su propia fecha. Turno noche (N,
   349→   349→  // 23:00 -> 06:00): la jornada que ANTECEDE al feriado es la feriada, así que
   350→   350→  // el op-día N con fecha D es feriado si D+1 lo es (los bultos de la noche
   351→   351→  // previa al feriado quedan registrados con la fecha del día previo).
   352→   352→  const MS_DIA = 86_400_000
   353→   353→  const feriadoRefDe = (fecha: Date, turno: string): string | null => {
   354→   354→    const k = dia(turno === 'N' ? new Date(fecha.getTime() + MS_DIA) : fecha)
   355→   355→    return feriadosMap.has(k) ? k : null
   356→   356→  }
   357→   357→
   358→   358→  // --- serie diaria (cada día separa la medición normal de la producción feriada) ---
   359→   359→  type DiaAgg = {
   360→   360→    bultos: number; bultosBase: number; bultosExtras: number; bultosFeriado: number
   361→   361→    horas: number; horasNormales: number; horasExtras: number; horasFeriado: number
   362→   362→    ops: Set<string>; opsNormales: Set<string>; opsExtras: Set<string>; opsFeriado: Set<string>
   363→   363→    opDias: number; opDiasNormales: number; opDiasFeriado: number
   364→   364→  }
   365→   365→  const porDia = new Map<string, DiaAgg>()
   366→   366→  for (const r of ops) {
   367→   367→    const k = dia(r.fecha)
   368→   368→    let p = porDia.get(k)
   369→   369→    if (!p) {
   370→   370→      p = { bultos: 0, bultosBase: 0, bultosExtras: 0, bultosFeriado: 0, horas: 0, horasNormales: 0, horasExtras: 0, horasFeriado: 0, ops: new Set(), opsNormales: new Set(), opsExtras: new Set(), opsFeriado: new Set(), opDias: 0, opDiasNormales: 0, opDiasFeriado: 0 }
   371→   371→      porDia.set(k, p)
   372→   372→    }
   373→   373→    p.bultos += r.bultos
   374→   374→    p.horas += r.horasActivas
   375→   375→    p.ops.add(r.operario)
   376→   376→    p.opDias += 1
   377→   377→    if (feriadoRefDe(r.fecha, r.turno)) {
   378→   378→      p.bultosFeriado += r.bultos
   379→   379→      p.horasFeriado += r.horasActivas
   380→   380→      p.opsFeriado.add(r.operario)
   381→   381→      p.opDiasFeriado += 1
   382→   382→    } else {
   383→   383→      p.bultosBase += r.bultosBase
   384→   384→      p.bultosExtras += r.bultosExtras
   385→   385→      p.horasNormales += r.horasActivas
   386→   386→      p.horasExtras += r.extras
   387→   387→      p.opsNormales.add(r.operario)
   388→   388→      if (r.extras > 0) p.opsExtras.add(r.operario)
   389→   389→      p.opDiasNormales += 1
   390→   390→    }
   391→   391→  }
   392→   392→
   393→   393→  const serie = [...porDia.entries()].sort().map(([fecha, p]) => {
   394→   394→    const bultosNormales = p.bultos - p.bultosFeriado
   395→   395→    const manana = feriadosMap.get(dia(new Date(new Date(`${fecha}T00:00:00.000Z`).getTime() + MS_DIA)))?.nombre ?? null
   396→   396→    return {
   397→   397→      fecha,
   398→   398→      bultos: p.bultos,
   399→   399→      bultosNormales,
   400→   400→      bultosBase: p.bultosBase,
   401→   401→      bultosExtras: p.bultosExtras,
   402→   402→      bultosFeriado: p.bultosFeriado,
   403→   403→      pctExtras: bultosNormales ? +((p.bultosExtras / bultosNormales) * 100).toFixed(1) : 0,
   404→   404→      horas: p.horas,
   405→   405→      horasNormales: p.horasNormales,
   406→   406→      horasExtras: p.horasExtras,
   407→   407→      horasFeriado: p.horasFeriado,
   408→   408→      ritmo: p.horasNormales ? +(bultosNormales / p.horasNormales).toFixed(1) : null,
   409→   409→      ops: p.ops.size,
   410→   410→      opsExtras: p.opsExtras.size,
   411→   411→      opsFeriado: p.opsFeriado.size,
   412→   412→      opDias: p.opDias,
   413→   413→      opDiasNormales: p.opDiasNormales,
   414→   414→      opDiasFeriado: p.opDiasFeriado,
   415→   415→      esFeriado: feriadosMap.has(fecha),
   416→   416→      feriado: feriadosMap.get(fecha)?.nombre ?? null,
   417→   417→      feriadoManana: manana,
   418→   418→    }
   419→   419→  })
   420→   420→
   421→   421→  // --- agregados por mes ---
   422→   422→  // Medición normal: por fecha del op-día. La producción feriada se acumula en
   423→   423→  // el MES DEL FERIADO al que pertenece (para el TN puede ser el mes siguiente:
   424→   424→  // la noche del 30/04 hacia el feriado del 01/05 cuenta como mayo).
   425→   425→  const mesAgg = new Map<string, { bultos: number; bultosBase: number; bultosExtras: number; horas: number; horasExtras: number; opDias: number; ops: Set<string>; opsExtras: Set<string>; ritmos: number[]; dias: number }>()
   426→   426→  for (const s of serie) {
   427→   427→    if (s.opDiasNormales === 0) continue
   428→   428→    const k = s.fecha.slice(0, 7)
   429→   429→    let m = mesAgg.get(k)
   430→   430→    if (!m) { m = { bultos: 0, bultosBase: 0, bultosExtras: 0, horas: 0, horasExtras: 0, opDias: 0, ops: new Set(), opsExtras: new Set(), ritmos: [], dias: 0 }; mesAgg.set(k, m) }
   431→   431→    m.dias += 1
   432→   432→    m.bultos += s.bultosNormales
   433→   433→    m.bultosBase += s.bultosBase
   434→   434→    m.bultosExtras += s.bultosExtras
   435→   435→    m.horas += s.horasNormales
   436→   436→    m.horasExtras += s.horasExtras
   437→   437→    m.opDias += s.opDiasNormales
   438→   438→    m.ritmos.push(s.ritmo ?? 0)
   439→   439→  }
   440→   440→  // producción feriada por mes del feriado (incluye la noche TN del día previo)
   441→   441→  const ferMes = new Map<string, { bultosFeriado: number; horasFeriado: number; refs: Set<string> }>()
   442→   442→  for (const r of ops) {
   443→   443→    const ref = feriadoRefDe(r.fecha, r.turno)
   444→   444→    if (!ref) continue
   445→   445→    const k = ref.slice(0, 7)
   446→   446→    let fm = ferMes.get(k)
   447→   447→    if (!fm) { fm = { bultosFeriado: 0, horasFeriado: 0, refs: new Set() }; ferMes.set(k, fm) }
   448→   448→    fm.bultosFeriado += r.bultos
   449→   449→    fm.horasFeriado += r.horasActivas
   450→   450→    fm.refs.add(ref)
   451→   451→  }
   452→   452→  // personas distintas por mes (necesita operarios por día, separando feriado)
   453→   453→  const opsPorDia = new Map<string, { opsNormales: Set<string>; opsExtras: Set<string>; opsFeriado: Set<string> }>()
   454→   454→  for (const r of ops) {
   455→   455→    const k = dia(r.fecha)
   456→   456→    let p = opsPorDia.get(k)
   457→   457→    if (!p) { p = { opsNormales: new Set(), opsExtras: new Set(), opsFeriado: new Set() }; opsPorDia.set(k, p) }
   458→   458→    if (feriadoRefDe(r.fecha, r.turno)) {
   459→   459→      p.opsFeriado.add(r.operario)
   460→   460→    } else {
   461→   461→      p.opsNormales.add(r.operario)
   462→   462→      if (r.extras > 0) p.opsExtras.add(r.operario)
   463→   463→    }
   464→   464→  }
   465→   465→  for (const s of serie) {
   466→   466→    if (s.opDiasNormales === 0) continue
   467→   467→    const m = mesAgg.get(s.fecha.slice(0, 7))
   468→   468→    const p = opsPorDia.get(s.fecha)
   469→   469→    if (!m || !p) continue
   470→   470→    for (const op of p.opsNormales) m.ops.add(op)
   471→   471→    for (const op of p.opsExtras) m.opsExtras.add(op)
   472→   472→  }
   473→   473→  const porMes = [...new Set([...mesAgg.keys(), ...ferMes.keys()])].sort().map((mes) => {
   474→   474→    const m = mesAgg.get(mes)
   475→   475→    const fm = ferMes.get(mes)
   476→   476→    const bultos = m?.bultos ?? 0
   477→   477→    const bultosExtras = m?.bultosExtras ?? 0
   478→   478→    const bultosFeriado = fm?.bultosFeriado ?? 0
   479→   479→    return {
   480→   480→      mes,
   481→   481→      dias: m?.dias ?? 0,
   482→   482→      diasFeriado: fm?.refs.size ?? 0,
   483→   483→      actividades: m?.opDias ?? 0,
   484→   484→      personas: m?.ops.size ?? 0,
   485→   485→      personasExtras: m?.opsExtras.size ?? 0,
   486→   486→      bultos,
   487→   487→      bultosBase: m?.bultosBase ?? 0,
   488→   488→      bultosExtras,
   489→   489→      bultosFeriado,
   490→   490→      pctExtras: bultos ? +((bultosExtras / bultos) * 100).toFixed(1) : 0,
   491→   491→      pctExtrasTotal: bultos + bultosFeriado ? +(((bultosExtras + bultosFeriado) / (bultos + bultosFeriado)) * 100).toFixed(1) : 0,
   492→   492→      horas: m?.horas ?? 0,
   493→   493→      horasExtras: m?.horasExtras ?? 0,
   494→   494→      horasFeriado: fm?.horasFeriado ?? 0,
   495→   495→      ritmoProm: m?.horas ? +(bultos / m.horas).toFixed(1) : null,
   496→   496→      ritmoMediana: m?.ritmos.length ? +mediana(m.ritmos).toFixed(1) : null,
   497→   497→    }
   498→   498→  })
   499→   499→
   500→   500→  // --- perfil por hora: colaboradores en jornada vs extras (promedio por dia) ---
   501→   501→  type HoraAgg = { fechas: Set<string>; jornada: Map<string, Set<string>>; extras: Map<string, Set<string>>; bultos: Map<string, number> }
   502→   502→  const horasAgg: HoraAgg[] = Array.from({ length: 24 }, () => ({ fechas: new Set<string>(), jornada: new Map(), extras: new Map(), bultos: new Map() }))
   503→   503→  for (const r of oh) {
   504→   504→    const h = horasAgg[r.hora]
   505→   505→    if (!h) continue
   506→   506→    const k = dia(r.fecha)
   507→   507→    // op-día feriado (feriado propio o noche TN previa al feriado): medición aparte
   508→   508→    const ref = r.turno === 'N' ? dia(new Date(r.fecha.getTime() + MS_DIA)) : k
   509→   509→    if (feriadosMap.has(ref)) continue
   510→   510→    h.fechas.add(k)
   511→   511→    const destino = r.esExtra ? h.extras : h.jornada
   512→   512→    let set = destino.get(k)
   513→   513→    if (!set) { set = new Set(); destino.set(k, set) }
   514→   514→    set.add(r.operario)
   515→   515→    h.bultos.set(k, (h.bultos.get(k) ?? 0) + r.bultos)
   516→   516→  }
   517→   517→  const perfilHora = horasAgg.map((h, hora) => {
   518→   518→    const n = h.fechas.size
   519→   519→    const sumaJornada = [...h.jornada.values()].reduce((a, s) => a + s.size, 0)
   520→   520→    const sumaExtras = [...h.extras.values()].reduce((a, s) => a + s.size, 0)
   521→   521→    const sumaBultos = [...h.bultos.values()].reduce((a, b) => a + b, 0)
   522→   522→    return {
   523→   523→      hora,
   524→   524→      etiqueta: `${String(hora).padStart(2, '0')}:00`,
   525→   525→      opsJornada: n ? +(sumaJornada / n).toFixed(1) : 0,
   526→   526→      opsExtras: n ? +(sumaExtras / n).toFixed(1) : 0,
   527→   527→      bultosProm: n ? Math.round(sumaBultos / n) : 0,
   528→   528→    }
   529→   529→  })
   530→   530→
   531→   531→  // --- resumen por turno (ventanas: TM 6-14, TT 14-22, TN 23-06) ---
   532→   532→  const turnoAgg = new Map<string, { bultos: number; bultosBase: number; bultosExtras: number; horas: number; horasExtras: number; opDias: number; ops: Set<string>; opsExtras: Set<string> }>()
   533→   533→  for (const r of ops) {
   534→   534→    if (feriadoRefDe(r.fecha, r.turno)) continue // jornadas feriadas se miden aparte
   535→   535→    let t = turnoAgg.get(r.turno)
   536→   536→    if (!t) { t = { bultos: 0, bultosBase: 0, bultosExtras: 0, horas: 0, horasExtras: 0, opDias: 0, ops: new Set(), opsExtras: new Set() }; turnoAgg.set(r.turno, t) }
   537→   537→    t.bultos += r.bultos
   538→   538→    t.bultosBase += r.bultosBase
   539→   539→    t.bultosExtras += r.bultosExtras
   540→   540→    t.horas += r.horasActivas
   541→   541→    t.horasExtras += r.extras
   542→   542→    t.opDias += 1
   543→   543→    t.ops.add(r.operario)
   544→   544→    if (r.extras > 0) t.opsExtras.add(r.operario)
   545→   545→  }
   546→   546→  const NOMBRE_TURNO: Record<string, string> = { M: 'TM (6 a 14)', T: 'TT (14 a 22)', N: 'TN (23 a 06)' }
   547→   547→  const porTurno = [...turnoAgg.entries()].sort().map(([turno, t]) => ({
   548→   548→    turno,
   549→   549→    nombre: NOMBRE_TURNO[turno] ?? turno,
   550→   550→    opDias: t.opDias,
   551→   551→    personas: t.ops.size,
   552→   552→    personasExtras: t.opsExtras.size,
   553→   553→    bultos: t.bultos,
   554→   554→    bultosBase: t.bultosBase,
   555→   555→    bultosExtras: t.bultosExtras,
   556→   556→    pctExtras: t.bultos ? +((t.bultosExtras / t.bultos) * 100).toFixed(1) : 0,
   557→   557→    horasExtras: t.horasExtras,
   558→   558→    ritmoProm: t.horas ? +(t.bultos / t.horas).toFixed(1) : null,
   559→   559→  }))
   560→   560→
   561→   561→  // --- feriados con actividad (se informan aparte) ---
   562→   562→  // Cada feriado reúne su jornada completa: los diurnos del propio feriado más
   563→   563→  // la noche del turno noche que lo antecede (23:00 -> 06:00 del día previo).
   564→   564→  const ferAgg = new Map<string, { bultos: number; horas: number; ops: Set<string>; opDias: number }>()
   565→   565→  for (const r of ops) {
   566→   566→    const ref = feriadoRefDe(r.fecha, r.turno)
   567→   567→    if (!ref) continue
   568→   568→    let v = ferAgg.get(ref)
   569→   569→    if (!v) { v = { bultos: 0, horas: 0, ops: new Set(), opDias: 0 }; ferAgg.set(ref, v) }
   570→   570→    v.bultos += r.bultos
   571→   571→    v.horas += r.horasActivas
   572→   572→    v.ops.add(r.operario)
   573→   573→    v.opDias += 1
   574→   574→  }
   575→   575→  const feriadosDetalle = [...ferAgg.entries()].sort().map(([fecha, v]) => ({
   576→   576→    fecha,
   577→   577→    nombre: feriadosMap.get(fecha)?.nombre ?? 'Feriado',
   578→   578→    tipo: feriadosMap.get(fecha)?.tipo ?? 'inamovible',
   579→   579→    personas: v.ops.size,
   580→   580→    opDias: v.opDias,
   581→   581→    bultos: v.bultos,
   582→   582→    horas: v.horas,
   583→   583→    ritmo: v.horas ? +(v.bultos / v.horas).toFixed(1) : null,
   584→   584→  }))
   585→   585→
   586→   586→  // --- resumen del periodo filtrado (medición normal = solo op-días no feriados) ---
   587→   587→  const normales = serie.filter((s) => s.opDiasNormales > 0)
   588→   588→  const bultos = normales.reduce((a, s) => a + s.bultosNormales, 0)
   589→   589→  const bultosBase = normales.reduce((a, s) => a + s.bultosBase, 0)
   590→   590→  const bultosExtras = normales.reduce((a, s) => a + s.bultosExtras, 0)
   591→   591→  const horas = normales.reduce((a, s) => a + s.horasNormales, 0)
   592→   592→  const horasExtras = normales.reduce((a, s) => a + s.horasExtras, 0)
   593→   593→  const bultosFeriado = [...ferAgg.values()].reduce((a, v) => a + v.bultos, 0)
   594→   594→  const horasFeriado = [...ferAgg.values()].reduce((a, v) => a + v.horas, 0)
   595→   595→  const personas = new Set<string>()
   596→   596→  const personasExtras = new Set<string>()
   597→   597→  const personasFeriado = new Set<string>()
   598→   598→  for (const p of opsPorDia.values()) {
   599→   599→    for (const op of p.opsNormales) personas.add(op)
   600→   600→    for (const op of p.opsExtras) personasExtras.add(op)
   601→   601→  }
   602→   602→  for (const v of ferAgg.values()) for (const op of v.ops) personasFeriado.add(op)
   603→   603→  const resumen = {
   604→   604→    dias: normales.length,
   605→   605→    actividades: normales.reduce((a, s) => a + s.opDiasNormales, 0),
   606→   606→    personas: personas.size,
   607→   607→    personasExtras: personasExtras.size,
   608→   608→    bultos,
   609→   609→    bultosBase,
   610→   610→    bultosExtras,
   611→   611→    pctExtras: bultos ? +((bultosExtras / bultos) * 100).toFixed(1) : 0,
   612→   612→    horas,
   613→   613→    horasExtras,
   614→   614→    ritmoProm: horas ? +(bultos / horas).toFixed(1) : null,
   615→   615→    ritmoMediana: +mediana(normales.map((s) => s.ritmo ?? 0)).toFixed(1),
   616→   616→    // feriados: medición aparte; sus bultos cuentan como extras en el total
   617→   617→    diasFeriado: ferAgg.size,
   618→   618→    bultosFeriado,
   619→   619→    horasFeriado,
   620→   620→    personasFeriado: personasFeriado.size,
   621→   621→    pctExtrasTotal: bultos + bultosFeriado ? +(((bultosExtras + bultosFeriado) / (bultos + bultosFeriado)) * 100).toFixed(1) : 0,
   622→   622→  }
   623→   623→
   624→   624→  return { serie, porMes, porTurno, perfilHora, resumen, feriados: feriadosDetalle, tieneOpHora: oh.length > 0 }
   625→   625→}

   628→   628→export async function getTM(f: Filtros & { incluirBajas?: boolean }) {
   629→   629→  const where: Record<string, unknown> = { fecha: rango(f) }
   630→   630→  if (!f.incluirBajas) where.OR = [{ estado: null }, { estado: { not: 'B' } }]
   631→   631→  if (f.turno) where.turno = f.turno
   632→   632→
   633→   633→  const rows = await db.tiempoMuerto.findMany({ where })
   634→   634→
   635→   635→  const totalMin = rows.reduce((a, r) => a + r.minutosEf, 0)
   636→   636→  const cats = new Map<string, { minutos: number; registros: number }>()
   637→   637→  for (const r of rows) {
   638→   638→    let c = cats.get(r.categoria)
   639→   639→    if (!c) { c = { minutos: 0, registros: 0 }; cats.set(r.categoria, c) }
   640→   640→    c.minutos += r.minutosEf
   641→   641→    c.registros += 1
   642→   642→  }
   643→   643→  const porCategoria = [...cats.entries()].map(([categoria, v]) => ({ categoria, ...v, pct: totalMin ? +((v.minutos / totalMin) * 100).toFixed(1) : 0 })).sort((a, b) => b.minutos - a.minutos)
   644→   644→
   645→   645→  // nave x pasillo (ESPERA UBICACION)
   646→   646→  const naves = new Map<string, { minutos: number; registros: number; pasillos: Map<string, { minutos: number; registros: number }> }>()
   647→   647→  for (const r of rows) {
   648→   648→    if (r.categoria !== 'ESPERA UBICACION' || !r.nave) continue
   649→   649→    let n = naves.get(r.nave)
   650→   650→    if (!n) { n = { minutos: 0, registros: 0, pasillos: new Map() }; naves.set(r.nave, n) }
   651→   651→    n.minutos += r.minutosEf
   652→   652→    n.registros += 1
   653→   653→    const pk = r.pasillo ?? '?'
   654→   654→    let p = n.pasillos.get(pk)
   655→   655→    if (!p) { p = { minutos: 0, registros: 0 }; n.pasillos.set(pk, p) }
   656→   656→    p.minutos += r.minutosEf
   657→   657→    p.registros += 1
   658→   658→  }
   659→   659→  const porNave = [...naves.entries()].map(([nave, v]) => ({
   660→   660→    nave,
   661→   661→    minutos: v.minutos,
   662→   662→    registros: v.registros,
   663→   663→    pasillos: [...v.pasillos.entries()].map(([pasillo, p]) => ({ pasillo, ...p })).sort((a, b) => b.minutos - a.minutos),
   664→   664→  })).sort((a, b) => b.minutos - a.minutos)
   665→   665→
   666→   666→  // por turno y por dia
   667→   667→  const turnosMap = new Map<string, number>()
   668→   668→  const diasMap = new Map<string, number>()
   669→   669→  const horasMap = new Map<number, number>()
   670→   670→  const detalles = new Map<string, { minutos: number; registros: number }>()
   671→   671→  for (const r of rows) {
   672→   672→    turnosMap.set(r.turno, (turnosMap.get(r.turno) ?? 0) + r.minutosEf)
   673→   673→    const k = dia(r.fecha)
   674→   674→    diasMap.set(k, (diasMap.get(k) ?? 0) + r.minutosEf)
   675→   675→    if (r.horaDesde != null) horasMap.set(Math.floor(r.horaDesde / 60), (horasMap.get(Math.floor(r.horaDesde / 60)) ?? 0) + r.minutosEf)
   676→   676→    const dk = r.detalle ?? '(sin dato)'
   677→   677→    let d = detalles.get(dk)
   678→   678→    if (!d) { d = { minutos: 0, registros: 0 }; detalles.set(dk, d) }
   679→   679→    d.minutos += r.minutosEf
   680→   680→    d.registros += 1
   681→   681→  }
   682→   682→  const NOM_TURNO: Record<string, string> = { TM: 'Mañana', TT: 'Tarde', TN: 'Noche' }
   683→   683→  const porTurno = [...turnosMap.entries()].map(([t, minutos]) => ({ turno: t, nombre: NOM_TURNO[t] ?? t, minutos })).sort((a, b) => b.minutos - a.minutos)
   684→   684→  const porDia = [...diasMap.entries()].sort().map(([fecha, minutos]) => ({ fecha, minutos }))
   685→   685→  const porHora = Array.from({ length: 24 }, (_, h) => ({ hora: h, etiqueta: `${String(h).padStart(2, '0')}:00`, minutos: horasMap.get(h) ?? 0 }))
   686→   686→  const topDetalles = [...detalles.entries()].map(([detalle, v]) => ({ detalle, ...v })).sort((a, b) => b.minutos - a.minutos).slice(0, 25)
   687→   687→
   688→   688→  // cruces codigo x categoria
   689→   689→  const codCat = new Map<string, { code: number | null; categoria: string; minutos: number; registros: number }>()
   690→   690→  for (const r of rows) {
   691→   691→    const k = `${r.motivoCode ?? '-'}|${r.categoria}`
   692→   692→    let c = codCat.get(k)
   693→   693→    if (!c) { c = { code: r.motivoCode, categoria: r.categoria, minutos: 0, registros: 0 }; codCat.set(k, c) }
   694→   694→    c.minutos += r.minutosEf
   695→   695→    c.registros += 1
   696→   696→  }
   697→   697→  const codigoCategoria = [...codCat.values()].sort((a, b) => b.minutos - a.minutos).slice(0, 20)
   698→   698→
   699→   699→  return {
   700→   700→    totalMin,
   701→   701→    totalHoras: +(totalMin / 60).toFixed(1),
   702→   702→    registros: rows.length,
   703→   703→    porCategoria,
   704→   704→    porNave,
   705→   705→    porTurno,
   706→   706→    porDia,
   707→   707→    porHora,
   708→   708→    topDetalles,
   709→   709→    codigoCategoria,
   710→   710→    navesResumen: {
   711→   711→      minutos: porNave.reduce((a, n) => a + n.minutos, 0),
   712→   712→      naves: porNave.length,
   713→   713→      pasillos: porNave.reduce((a, n) => a + n.pasillos.length, 0),
   714→   714→    },
   715→   715→  }
   716→   716→}

   719→   719→export async function getPicking(f: Filtros) {
   720→   720→  const where = { fecha: rango(f) }
   721→   721→  const [count, ops] = await Promise.all([
   722→   722→    db.pickingEvento.count({ where }),
   723→   723→    db.pickingEvento.findMany({ where, orderBy: [{ fecha: 'asc' }, { operario: 'asc' }, { horaMin: 'asc' }], select: { fecha: true, operario: true, horaMin: true, bultos: true, soporte: true, circuito: true } }),
   724→   724→  ])
   725→   725→
   726→   726→  if (!count) {
   727→   727→    return { registros: 0, vacio: true as const }
   728→   728→  }
   729→   729→
   730→   730→  // gaps entre eventos consecutivos por operario y dia
   731→   731→  const buckets = [
   732→   732→    { label: '0-1 min', min: 0, max: 1 },
   733→   733→    { label: '1-2 min', min: 1, max: 2 },
   734→   734→    { label: '2-5 min', min: 2, max: 5 },
   735→   735→    { label: '5-10 min', min: 5, max: 10 },
   736→   736→    { label: '10-20 min', min: 10, max: 20 },
   737→   737→    { label: '20-60 min', min: 20, max: 60 },
   738→   738→    { label: '> 60 min', min: 60, max: Infinity },
   739→   739→  ]
   740→   740→  const gapCounts = new Array(buckets.length).fill(0)
   741→   741→  const soporteGapCounts = new Array(buckets.length).fill(0)
   742→   742→  let gapSum = 0, gapN = 0, soporteGapSum = 0, soporteGapN = 0, cambiosSoporte = 0
   743→   743→  const opAgg = new Map<string, { nombre: string; eventos: number; gapSum: number; gapN: number; cambioSop: number; bultos: number }>()
   744→   744→  const serDia = new Map<string, { eventos: number; gapMin: number }>()
   745→   745→
   746→   746→  let prev: { operario: string; key: string; horaMin: number; soporte: string | null } | null = null
   747→   747→  for (const r of ops) {
   748→   748→    const opKey = r.operario
   749→   749→    let o = opAgg.get(opKey)
   750→   750→    if (!o) { o = { nombre: opKey, eventos: 0, gapSum: 0, gapN: 0, cambioSop: 0, bultos: 0 }; opAgg.set(opKey, o) }
   751→   751→    o.eventos += 1
   752→   752→    o.bultos += r.bultos ?? 0
   753→   753→    const k = r.fecha.toISOString().slice(0, 10)
   754→   754→    let s = serDia.get(k)
   755→   755→    if (!s) { s = { eventos: 0, gapMin: 0 }; serDia.set(k, s) }
   756→   756→    s.eventos += 1
   757→   757→
   758→   758→    if (prev && prev.operario === opKey && prev.key === k && r.horaMin != null && prev.horaMin != null) {
   759→   759→      const gap = Math.max(0, r.horaMin - prev.horaMin)
   760→   760→      if (gap > 720) {
   761→   761→        // salto de jornada (>12 h): no es tiempo muerto, se ignora para las metricas
   762→   762→        prev = { operario: opKey, key: k, horaMin: r.horaMin ?? -1, soporte: r.soporte }
   763→   763→        continue
   764→   764→      }
   765→   765→      const bi = buckets.findIndex((b) => gap >= b.min && gap < b.max)
   766→   766→      if (bi >= 0) gapCounts[bi]++
   767→   767→      gapSum += gap; gapN++
   768→   768→      s.gapMin += gap
   769→   769→      o.gapSum += gap; o.gapN++
   770→   770→      const cambioSop = prev.soporte != null && r.soporte != null && prev.soporte !== r.soporte
   771→   771→      if (cambioSop) {
   772→   772→        cambiosSoporte++
   773→   773→        if (bi >= 0) soporteGapCounts[bi]++
   774→   774→        soporteGapSum += gap; soporteGapN++
   775→   775→        o.cambioSop++
   776→   776→      }
   777→   777→    }
   778→   778→    prev = { operario: opKey, key: k, horaMin: r.horaMin ?? -1, soporte: r.soporte }
   779→   779→  }
   780→   780→
   781→   781→  const serie = [...serDia.entries()].sort().map(([fecha, v]) => ({ fecha, eventos: v.eventos, gapPromedio: v.eventos > 1 ? +(v.gapMin / (v.eventos - 1)).toFixed(2) : null }))
   782→   782→
   783→   783→  return {
   784→   784→    registros: count,
   785→   785→    vacio: false as const,
   786→   786→    operarios: opAgg.size,
   787→   787→    gapPromedio: gapN ? +(gapSum / gapN).toFixed(2) : null,
   788→   788→    gapMedianaEst: gapN ? +(gapSum / gapN).toFixed(2) : null,
   789→   789→    cambiosSoporte,
   790→   790→    gapPromedioCambioSoporte: soporteGapN ? +(soporteGapSum / soporteGapN).toFixed(2) : null,
   791→   791→    distribucionGaps: buckets.map((b, i) => ({ bucket: b.label, cantidad: gapCounts[i] })),
   792→   792→    distribucionGapsSoporte: buckets.map((b, i) => ({ bucket: b.label, cantidad: soporteGapCounts[i] })),
   793→   793→    operariosTop: [...opAgg.values()].filter((o) => o.gapN > 30).map((o) => ({ operario: o.nombre, eventos: o.eventos, bultos: o.bultos, gapPromedio: +(o.gapSum / o.gapN).toFixed(2), cambiosSoporte: o.cambioSop })).sort((a, b) => b.eventos - a.eventos).slice(0, 15),
   794→   794→    serie,
   795→   795→  }
   796→   796→}

   799→   799→const r1 = (x: number) => Math.round(x * 10) / 10

   802→   802→export async function getProdCirc(f: Filtros) {
   803→   803→  const where = { fecha: rango(f) }
   804→   804→  const [tot, sectores, filas, ops] = await Promise.all([
   805→   805→    db.prodCircuito.aggregate({
   806→   806→      where,
   807→   807→      _count: true,
   808→   808→      _min: { fecha: true },
   809→   809→      _max: { fecha: true },
   810→   810→      _sum: { soportes: true, lineas: true, bultos: true, tiempoTotal: true, tiempoMuerto: true, tiempoNeto: true, tiempoSuperNeto: true },
   811→   811→    }),
   812→   812→    db.prodCircuito.groupBy({
   813→   813→      by: ['sector'],
   814→   814→      where,
   815→   815→      _sum: { soportes: true, bultos: true, tiempoTotal: true, tiempoMuerto: true, tiempoSuperNeto: true },
   816→   816→    }),
   817→   817→    db.prodCircuito.findMany({ where, select: { fecha: true, bultos: true, tiempoTotal: true, tiempoMuerto: true, tiempoSuperNeto: true } }),
   818→   818→    db.prodCircuito.groupBy({
   819→   819→      by: ['operario', 'nombre'],
   820→   820→      where,
   821→   821→      _sum: { bultos: true, soportes: true, tiempoTotal: true, tiempoMuerto: true, tiempoSuperNeto: true },
   822→   822→    }),
   823→   823→  ])
   824→   824→
   825→   825→  if (!tot._count) return { registros: 0, vacio: true as const }
   826→   826→
   827→   827→  const s = tot._sum
   828→   828→  const kpis = {
   829→   829→    registros: tot._count,
   830→   830→    soportes: Math.round(s.soportes ?? 0),
   831→   831→    lineas: Math.round(s.lineas ?? 0),
   832→   832→    bultos: Math.round(s.bultos ?? 0),
   833→   833→    horasTotal: r1(s.tiempoTotal ?? 0),
   834→   834→    horasMuerto: r1(s.tiempoMuerto ?? 0),
   835→   835→    horasNeto: r1(s.tiempoNeto ?? 0),
   836→   836→    pctMuerto: s.tiempoTotal ? r1(((s.tiempoMuerto ?? 0) / s.tiempoTotal) * 100) : null,
   837→   837→    prodTotal: s.tiempoTotal ? r2((s.bultos ?? 0) / s.tiempoTotal) : null,
   838→   838→    prodNeto: s.tiempoNeto ? r2((s.bultos ?? 0) / s.tiempoNeto) : null,
   839→   839→    prodSuperNeto: s.tiempoSuperNeto ? r2((s.bultos ?? 0) / s.tiempoSuperNeto) : null,
   840→   840→  }
   841→   841→
   842→   842→  const porSector = sectores
   843→   843→    .map((x) => ({
   844→   844→      sector: x.sector || '(sin sector)',
   845→   845→      soportes: Math.round(x._sum.soportes ?? 0),
   846→   846→      bultos: Math.round(x._sum.bultos ?? 0),
   847→   847→      pctMuerto: x._sum.tiempoTotal ? r1(((x._sum.tiempoMuerto ?? 0) / x._sum.tiempoTotal) * 100) : null,
   848→   848→      prodSuperNeto: x._sum.tiempoSuperNeto && x._sum.tiempoSuperNeto > 0 ? r2((x._sum.bultos ?? 0) / x._sum.tiempoSuperNeto) : null,
   849→   849→    }))
   850→   850→    .sort((a, b) => b.bultos - a.bultos)
   851→   851→
   852→   852→  // serie mensual
   853→   853→  const porMesMap = new Map<string, { bultos: number; tiempoTotal: number; tiempoMuerto: number; tiempoSuperNeto: number }>()
   854→   854→  for (const r of filas) {
   855→   855→    const k = dia(r.fecha).slice(0, 7)
   856→   856→    let m = porMesMap.get(k)
   857→   857→    if (!m) { m = { bultos: 0, tiempoTotal: 0, tiempoMuerto: 0, tiempoSuperNeto: 0 }; porMesMap.set(k, m) }
   858→   858→    m.bultos += r.bultos ?? 0
   859→   859→    m.tiempoTotal += r.tiempoTotal ?? 0
   860→   860→    m.tiempoMuerto += r.tiempoMuerto ?? 0
   861→   861→    m.tiempoSuperNeto += r.tiempoSuperNeto ?? 0
   862→   862→  }
   863→   863→  const porMes = [...porMesMap.entries()].sort(([a], [b]) => a.localeCompare(b)).map(([mes, m]) => ({
   864→   864→    mes,
   865→   865→    bultos: Math.round(m.bultos),
   866→   866→    pctMuerto: m.tiempoTotal ? r1((m.tiempoMuerto / m.tiempoTotal) * 100) : null,
   867→   867→    prodSuperNeto: m.tiempoSuperNeto > 0 ? r2(m.bultos / m.tiempoSuperNeto) : null,
   868→   868→  }))
   869→   869→
   870→   870→  const operariosTop = ops
   871→   871→    .map((x) => ({
   872→   872→      operario: x.operario,
   873→   873→      nombre: x.nombre || x.operario,
   874→   874→      bultos: Math.round(x._sum.bultos ?? 0),
   875→   875→      soportes: Math.round(x._sum.soportes ?? 0),
   876→   876→      horasTotal: r1(x._sum.tiempoTotal ?? 0),
   877→   877→      pctMuerto: x._sum.tiempoTotal ? r1(((x._sum.tiempoMuerto ?? 0) / x._sum.tiempoTotal) * 100) : null,
   878→   878→      prodSuperNeto: x._sum.tiempoSuperNeto && x._sum.tiempoSuperNeto > 0 ? r2((x._sum.bultos ?? 0) / x._sum.tiempoSuperNeto) : null,
   879→   879→    }))
   880→   880→    .sort((a, b) => b.bultos - a.bultos)
   881→   881→    .slice(0, 15)
   882→   882→
   883→   883→  return {
   884→   884→    registros: tot._count,
   885→   885→    vacio: false as const,
   886→   886→    desde: tot._min.fecha ? dia(tot._min.fecha) : undefined,
   887→   887→    hasta: tot._max.fecha ? dia(tot._max.fecha) : undefined,
   888→   888→    kpis,
   889→   889→    porSector,
   890→   890→    porMes,
   891→   891→    operariosTop,
   892→   892→  }
   893→   893→}
