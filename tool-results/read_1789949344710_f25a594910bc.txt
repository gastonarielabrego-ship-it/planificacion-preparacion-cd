    80→    80→// Caja de comparación: valor de un día normal contra el de una jornada feriada
    81→    81→function CajaComparativa({ titulo, normal, feriado, delta, unidad, decimales }: {
    82→    82→  titulo: string
    83→    83→  normal: number | null
    84→    84→  feriado: number | null
    85→    85→  delta: number | null
    86→    86→  unidad?: string
    87→    87→  decimales?: boolean
    88→    88→}) {
    89→    89→  const fmtV = (v: number | null) => (decimales ? n1(v) : n(v))
    90→    90→  return (
    91→    91→    <div className="rounded-lg border p-3 space-y-2">
    92→    92→      <p className="text-xs text-muted-foreground">{titulo}</p>
    93→    93→      <div className="flex items-end gap-3">
    94→    94→        <div>
    95→    95→          <p className="text-[11px] font-medium text-emerald-700">Día normal</p>
    96→    96→          <p className="text-xl font-bold tabular-nums">{fmtV(normal)}<span className="text-xs font-normal text-muted-foreground"> {unidad}</span></p>
    97→    97→        </div>
    98→    98→        <div className="border-l pl-3">
    99→    99→          <p className="text-[11px] font-medium text-red-700">Feriado</p>
   100→   100→          <p className="text-xl font-bold tabular-nums text-red-700">{fmtV(feriado)}<span className="text-xs font-normal text-muted-foreground"> {unidad}</span></p>
   101→   101→        </div>
   102→   102→      </div>
   103→   103→      {delta !== null && (
   104→   104→        <Badge variant="outline" className={delta >= 0 ? 'border-emerald-200 bg-emerald-50 text-emerald-700' : 'border-red-200 bg-red-50 text-red-700'}>
   105→   105→          {delta >= 0 ? '+' : ''}{n1(delta)}% vs día normal
   106→   106→        </Badge>
   107→   107→      )}
   108→   108→    </div>
   109→   109→  )
   110→   110→}
