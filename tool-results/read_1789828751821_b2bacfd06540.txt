    12→function rango(f: Filtros): { gte?: Date; lte?: Date } {
    13→  const r: { gte?: Date; lte?: Date } = {}
    14→  if (f.from) r.gte = new Date(f.from + 'T00:00:00.000Z')
    15→  if (f.to) r.lte = new Date(f.to + 'T23:59:59.999Z')
    16→  return r
    17→}
    18→const dia = (d: Date) => d.toISOString().slice(0, 10)

    20→// ============ RESUMEN EJECUTIVO ============
    21→export async function getResumen() {
    22→  const [olaCount, h61Count, tmCount, pickCount, batches] = await Promise.all([
    23→    db.olaDia.aggregate({ _count: true, _min: { fecha: true }, _max: { fecha: true }, _avg: { ola: true, total: true } }),
    24→    db.h61OpDia.aggregate({ _count: true, _min: { fecha: true }, _max: { fecha: true }, _sum: { bultos: true, horasActivas: true, extras: true, bultosExtras: true } }),
    25→    db.tiempoMuerto.aggregate({ _count: true, _min: { fecha: true }, _max: { fecha: true }, _sum: { minutos: true, minutosEf: true } }),
    26→    db.pickingEvento.aggregate({ _count: true, _min: { fecha: true }, _max: { fecha: true } }),
    27→    db.uploadBatch.findMany({ orderBy: { createdAt: 'desc' }, take: 10 }),
    28→  ])
    29→
    30→  const opsUnicos = await db.h61OpDia.findMany({ select: { operario: true }, distinct: ['operario'] })
    31→
    32→  // top categorias TM (excluye bajas ESTADO B; OR con null es clave: NOT excludes NULLs en SQL)
    33→  const tmCats = await db.tiempoMuerto.groupBy({
    34→    by: ['categoria'],
    35→    _sum: { minutosEf: true },
    36→    _count: true,
    37→    where: { OR: [{ estado: null }, { estado: { not: 'B' } }] },
    38→    orderBy: { _sum: { minutosEf: 'desc' } },
    39→  })
    40→
    41→  const horasTotales = h61Count._sum.horasActivas ?? 0
    42→  const bultos = h61Count._sum.bultos ?? 0
    43→  const bultosExtras = h61Count._sum.bultosExtras ?? 0
    44→  const totalMinTM = tmCats.reduce((a, c) => a + (c._sum.minutosEf ?? 0), 0)
    45→
    46→  return {
    47→    datasets: {
    48→      ola: { registros: olaCount._count, desde: olaCount._min.fecha, hasta: olaCount._max.fecha, olaMedia: Math.round(olaCount._avg.ola ?? 0), totalMedia: Math.round(olaCount._avg.total ?? 0) },
    49→      h61: { registros: h61Count._count, desde: h61Count._min.fecha, hasta: h61Count._max.fecha, bultos, horas: horasTotales, productividad: horasTotales ? +(bultos / horasTotales).toFixed(1) : 0, operarios: opsUnicos.length, extrasHoras: h61Count._sum.extras ?? 0, bultosExtras, pctBultosExtras: bultos ? +((bultosExtras / bultos) * 100).toFixed(1) : 0 },
    50→      tm: { registros: tmCount._count, desde: tmCount._min.fecha, hasta: tmCount._max.fecha, minutos: totalMinTM, horas: +(totalMinTM / 60).toFixed(0), pctSobreHorasH61: horasTotales ? +(((totalMinTM / 60) / horasTotales) * 100).toFixed(1) : 0 },
    51→      picking: { registros: pickCount._count, desde: pickCount._min.fecha, hasta: pickCount._max.fecha },
    52→    },
    53→    topCategoriasTM: tmCats.slice(0, 6).map((c) => ({ categoria: c.categoria, minutos: c._sum.minutosEf ?? 0, registros: c._count })),
    54→    batches: batches.map((b) => ({ ...b, createdAt: b.createdAt.toISOString() })),
    55→  }
    56→}

    58→// ============ OLA / PLANIFICACION ============
    59→export async function getPlanificacion(f: Filtros) {
    60→  const where = { fecha: rango(f) }
    61→  const [ola, h61] = await Promise.all([
    62→    db.olaDia.findMany({ where, orderBy: { fecha: 'asc' } }),
    63→    db.h61OpDia.findMany({ where: { fecha: rango(f) }, select: { fecha: true, operario: true, horasActivas: true, bultos: true, bultosBase: true, bultosExtras: true, extras: true, turno: true, funcion: true } }),
    64→  ])
    65→
    66→  const porDia = new Map<string, { ops: number; opsExtras: number; horas: number; horasExtras: number; bultos: number; bultosBase: number; bultosExtras: number }>()
    67→  for (const r of h61) {
    68→    const k = dia(r.fecha)
    69→    let p = porDia.get(k)
    70→    if (!p) { p = { ops: 0, opsExtras: 0, horas: 0, horasExtras: 0, bultos: 0, bultosBase: 0, bultosExtras: 0 }; porDia.set(k, p) }
    71→    p.ops += 1
    72→    if (r.extras > 0) p.opsExtras += 1
    73→    p.horas += r.horasActivas
    74→    p.horasExtras += r.extras
    75→    p.bultos += r.bultos
    76→    p.bultosBase += r.bultosBase
    77→    p.bultosExtras += r.bultosExtras
    78→  }
    79→
    80→  const fechas = new Set<string>([...ola.map((o) => dia(o.fecha)), ...porDia.keys()])
    81→  const serie = [...fechas].sort().map((k) => {
    82→    const o = ola.find((x) => dia(x.fecha) === k)
    83→    const p = porDia.get(k)
    84→    const demanda = o?.total ?? null
    85→    const prodHora = p && p.horas ? p.bultos / p.horas : null
    86→    // capacidad si se suprimen extras: mantiene bultosBase (produccion en <=8h por operario)
    87→    const faltanteSinExtras = p && demanda != null ? Math.max(0, demanda - p.bultosBase) : null
    88→    return {
    89→      fecha: k,
    90→      ola: o?.ola ?? null,
    91→      pendiente: o?.pendiente ?? null,
    92→      demanda,
    93→      preparado: p?.bultos ?? null,
    94→      ops: p?.ops ?? null,
    95→      opsExtras: p?.opsExtras ?? null,
    96→      horas: p ? +p.horas.toFixed(1) : null,
    97→      horasExtras: p?.horasExtras ?? 0,
    98→      prodHora: prodHora != null ? +prodHora.toFixed(1) : null,
    99→      bultosBase: p?.bultosBase ?? null,
   100→      bultosExtras: p?.bultosExtras ?? null,
   101→      faltanteSinExtras: faltanteSinExtras != null ? Math.round(faltanteSinExtras) : null,
   102→    }
   103→  })
   104→
   105→  // agregados por mes
   106→  const porMes = new Map<string, { ola: number; demanda: number; preparado: number; dias: number; diasExtras: number; horasExtras: number; faltante: number }>()
   107→  for (const s of serie) {
   108→    const k = s.fecha.slice(0, 7)
   109→    let m = porMes.get(k)
   110→    if (!m) { m = { ola: 0, demanda: 0, preparado: 0, dias: 0, diasExtras: 0, horasExtras: 0, faltante: 0 }; porMes.set(k, m) }
   111→    m.dias += 1
   112→    if (s.demanda != null) m.demanda += s.demanda
   113→    if (s.ola != null) m.ola += s.ola
   114→    if (s.preparado != null) m.preparado += s.preparado
   115→    if ((s.opsExtras ?? 0) > 0) m.diasExtras += 1
   116→    m.horasExtras += s.horasExtras ?? 0
   117→    if (s.faltanteSinExtras != null) m.faltante += s.faltanteSinExtras
   118→  }
   119→  const meses = [...porMes.entries()].sort().map(([mes, v]) => ({ mes, ...v }))
   120→
   121→  const conExtras = serie.filter((s) => (s.opsExtras ?? 0) > 0)
   122→  const resumen = {
   123→    dias: serie.length,
   124→    diasConExtras: conExtras.length,
   125→    pctDiasConExtras: serie.length ? +((conExtras.length / serie.length) * 100).toFixed(1) : 0,
   126→    faltantePromedioSinExtras: conExtras.length ? Math.round(conExtras.reduce((a, s) => a + (s.faltanteSinExtras ?? 0), 0) / conExtras.length) : 0,
   127→    faltanteTotalSinExtras: conExtras.reduce((a, s) => a + (s.faltanteSinExtras ?? 0), 0),
   128→    horasExtrasTotales: serie.reduce((a, s) => a + (s.horasExtras ?? 0), 0),
   129→    bultosExtrasTotales: serie.reduce((a, s) => a + (s.bultosExtras ?? 0), 0),
   130→    prodHoraMedia: (() => {
   131→      const vals = serie.filter((s) => s.prodHora != null && s.preparado != null && (s.preparado ?? 0) > 0)
   132→      const totB = vals.reduce((a, s) => a + (s.preparado ?? 0), 0)
   133→      const totH = vals.reduce((a, s) => a + (s.horas ?? 0), 0)
   134→      return totH ? +(totB / totH).toFixed(1) : null
   135→    })(),
   136→  }
   137→
   138→  return { serie, meses, resumen }
   139→}

   141→// ============ PRODUCTIVIDAD H61 ============
   142→export async function getH61(f: Filtros) {
   143→  const where: Record<string, unknown> = { fecha: rango(f) }
   144→  if (f.turno) where.turno = f.turno
   145→  if (f.funcion) where.funcion = f.funcion
   146→
   147→  const [ops, th, ci] = await Promise.all([
   148→    db.h61OpDia.findMany({ where }),
   149→    db.h61TurnoHora.findMany({ where: { fecha: rango(f) } }),
   150→    db.h61Circuito.findMany({ where: { fecha: rango(f) } }),
   151→  ])
   152→
   153→  // Perfil 24h: horas 0-5 del turno noche (N) pertenecen al dia calendario siguiente
   154→  const porHora = Array.from({ length: 24 }, () => ({ bultos: 0, operarios: new Set<string>() }))
   155→  const opSeen = new Map<string, Set<number>>() // fecha|op -> horas
   156→  for (const r of th) {
   157→    const fechaBase = new Date(r.fecha)
   158→    const esNocheMadrugada = r.turno === 'N' && r.hora <= 5
   159→    if (esNocheMadrugada) fechaBase.setUTCDate(fechaBase.getUTCDate() + 1)
   160→    const h = porHora[r.hora]
   161→    h.bultos += r.bultos
   162→    const k = `${dia(fechaBase)}|${r.turno}`
   163→    // operarios por hora aproximamos con el maximo informado ese dia-turno
   164→    const set = opSeen.get(k) ?? new Set<number>()
   165→    set.add(r.operarios)
   166→    opSeen.set(k, set)
   167→  }
   168→  const perfilHora = porHora.map((h, i) => ({
   169→    hora: i,
   170→    etiqueta: `${String(i).padStart(2, '0')}:00`,
   171→    bultos: h.bultos,
   172→    operariosProm: (() => {
   173→      const vals = [...opSeen.values()].map((s) => Math.max(...s))
   174→      return vals.length ? +(vals.reduce((a, b) => a + b, 0) / vals.length).toFixed(1) : 0
   175→    })(),
   176→  }))
   177→  const horasConDatos = perfilHora.filter((p) => p.bultos > 0)
   178→  const pico = horasConDatos.reduce((a, b) => (b.bultos > a.bultos ? b : a), horasConDatos[0] ?? { hora: 0, etiqueta: '-', bultos: 0, operariosProm: 0 })
   179→  const valle = horasConDatos.reduce((a, b) => (b.bultos < a.bultos ? b : a), horasConDatos[0] ?? { hora: 0, etiqueta: '-', bultos: 0, operariosProm: 0 })
   180→
   181→  // por turno
   182→  const turnoAgg = new Map<string, { bultos: number; horas: number; opsDias: number; bultosExtras: number }>()
   183→  for (const r of th) {
   184→    let t = turnoAgg.get(r.turno)
   185→    if (!t) { t = { bultos: 0, horas: 0, opsDias: 0, bultosExtras: 0 }; turnoAgg.set(r.turno, t) }
   186→    t.bultos += r.bultos
   187→  }
   188→  const turnoHoras = new Map<string, Set<string>>()
   189→  for (const r of ops) {
   190→    let t = turnoHoras.get(r.turno)
   191→    if (!t) { t = new Set(); turnoHoras.set(r.turno, t) }
   192→    t.add(`${dia(r.fecha)}|${r.operario}`)
   193→    t.add(`${dia(r.fecha)}|${r.operario}|h|${r.horasActivas}`)
   194→  }
   195→  for (const [turno, set] of turnoHoras) {
   196→    const horas = [...set].filter((x) => x.includes('|h|')).reduce((a, x) => a + parseInt(x.split('|h|')[1], 10), 0)
   197→    const opsDias = [...set].filter((x) => !x.includes('|h|')).length
   198→    const t = turnoAgg.get(turno)
   199→    if (t) { t.horas = horas; t.opsDias = opsDias }
   200→  }
   201→  for (const r of ops) {
   202→    const t = turnoAgg.get(r.turno)
   203→    if (t) t.bultosExtras += r.bultosExtras
   204→  }
   205→  const porTurno = [...turnoAgg.entries()].map(([turno, v]) => ({
   206→    turno,
   207→    nombre: turno === 'M' ? 'Mañana' : turno === 'T' ? 'Tarde' : turno === 'N' ? 'Noche' : turno,
   208→    ...v,
   209→    prodHora: v.horas ? +(v.bultos / v.horas).toFixed(1) : 0,
   210→  })).sort((a, b) => b.bultos - a.bultos)
   211→
   212→  // heatmap turno x hora
   213→  const heat = ['M', 'T', 'N'].map((turno) => {
   214→    const fila = Array.from({ length: 24 }, () => 0)
   215→    for (const r of th) if (r.turno === turno) fila[r.hora] = Math.max(fila[r.hora], Math.round(r.bultos / Math.max(1, r.operarios)))
   216→    return { turno, valores: fila }
   217→  })
   218→
   219→  // circuitos
   220→  const ciAgg = new Map<string, number>()
   221→  for (const r of ci) ciAgg.set(r.circuito, (ciAgg.get(r.circuito) ?? 0) + r.bultos)
   222→  const circuitos = [...ciAgg.entries()].map(([circuito, bultos]) => ({ circuito, bultos })).sort((a, b) => b.bultos - a.bultos)
   223→
   224→  // distribucion de jornadas 8h vs extras
   225→  const distHoras = new Map<number, number>()
   226→  for (const r of ops) distHoras.set(r.horasActivas, (distHoras.get(r.horasActivas) ?? 0) + 1)
   227→  const distribucion = [...distHoras.entries()].sort((a, b) => a[0] - b[0]).map(([horas, opsDia]) => ({ horas, opsDia }))
   228→
   229→  const opsExtras = ops.filter((r) => r.extras > 0)
   230→  const extrasResumen = {
   231→    opsDiasTotal: ops.length,
   232→    opsDiasConExtras: opsExtras.length,
   233→    pctOpsDiasConExtras: ops.length ? +((opsExtras.length / ops.length) * 100).toFixed(1) : 0,
   234→    horasExtras: opsExtras.reduce((a, r) => a + r.extras, 0),
   235→    bultosExtras: opsExtras.reduce((a, r) => a + r.bultosExtras, 0),
   236→    bultosTotales: ops.reduce((a, r) => a + r.bultos, 0),
   237→  }
   238→  extrasResumen.pctBultosExtras = extrasResumen.bultosTotales ? +((extrasResumen.bultosExtras / extrasResumen.bultosTotales) * 100).toFixed(1) : 0
   239→
   240→  // top operarios por productividad (minimo 80 horas)
   241→  const opAgg = new Map<string, { nombre: string | null; bultos: number; horas: number; extras: number; dias: number; funcion: string }>()
   242→  for (const r of ops) {
   243→    let o = opAgg.get(r.operario)
   244→    if (!o) { o = { nombre: r.nombre, bultos: 0, horas: 0, extras: 0, dias: 0, funcion: r.funcion }; opAgg.set(r.operario, o) }
   245→    if (!o.nombre && r.nombre) o.nombre = r.nombre
   246→    o.bultos += r.bultos
   247→    o.horas += r.horasActivas
   248→    o.extras += r.extras
   249→    o.dias += 1
   250→  }
   251→  const operarios = [...opAgg.entries()]
   252→    .map(([operario, o]) => ({ operario, nombre: o.nombre ?? operario, bultos: o.bultos, horas: o.horas, extras: o.extras, dias: o.dias, prodHora: o.horas ? +(o.bultos / o.horas).toFixed(1) : 0 }))
   253→    .filter((o) => o.horas >= 80)
   254→    .sort((a, b) => b.prodHora - a.prodHora)
   255→
   256→  // serie diaria
   257→  const serDia = new Map<string, { bultos: number; horas: number; ops: number }>()
   258→  for (const r of ops) {
   259→    const k = dia(r.fecha)
   260→    let s = serDia.get(k)
   261→    if (!s) { s = { bultos: 0, horas: 0, ops: 0 }; serDia.set(k, s) }
   262→    s.bultos += r.bultos
   263→    s.horas += r.horasActivas
   264→    s.ops += 1
   265→  }
   266→  const serie = [...serDia.entries()].sort().map(([fecha, v]) => ({ fecha, ...v, prodHora: v.horas ? +(v.bultos / v.horas).toFixed(1) : 0 }))
   267→
   268→  const bultosTot = ops.reduce((a, r) => a + r.bultos, 0)
   269→  const horasTot = ops.reduce((a, r) => a + r.horasActivas, 0)
   270→
   271→  return {
   272→    perfilHora,
   273→    pico: { ...pico },
   274→    valle: { ...valle },
   275→    porTurno,
   276→    heat,
   277→    circuitos: circuitos.slice(0, 17),
   278→    distribucion,
   279→    extrasResumen,
   280→    operariosTop: operarios.slice(0, 15),
   281→    operariosBottom: operarios.slice(-15).reverse(),
   282→    operariosExtras: [...opAgg.entries()].map(([operario, o]) => ({ operario, nombre: o.nombre ?? operario, extras: o.extras, dias: o.dias })).filter((o) => o.extras > 0).sort((a, b) => b.extras - a.extras).slice(0, 15),
   283→    serie,
   284→    resumen: { bultos: bultosTot, horas: horasTot, prodHora: horasTot ? +(bultosTot / horasTot).toFixed(1) : 0, opsUnicos: opAgg.size },
   285→  }
   286→}

   288→// ============ TIEMPOS MUERTOS ============
   289→export async function getTM(f: Filtros & { incluirBajas?: boolean }) {
   290→  const where: Record<string, unknown> = { fecha: rango(f) }
   291→  if (!f.incluirBajas) where.OR = [{ estado: null }, { estado: { not: 'B' } }]
   292→  if (f.turno) where.turno = f.turno
   293→
   294→  const rows = await db.tiempoMuerto.findMany({ where })
   295→
   296→  const totalMin = rows.reduce((a, r) => a + r.minutosEf, 0)
   297→  const cats = new Map<string, { minutos: number; registros: number }>()
   298→  for (const r of rows) {
   299→    let c = cats.get(r.categoria)
   300→    if (!c) { c = { minutos: 0, registros: 0 }; cats.set(r.categoria, c) }
   301→    c.minutos += r.minutosEf
   302→    c.registros += 1
   303→  }
   304→  const porCategoria = [...cats.entries()].map(([categoria, v]) => ({ categoria, ...v, pct: totalMin ? +((v.minutos / totalMin) * 100).toFixed(1) : 0 })).sort((a, b) => b.minutos - a.minutos)
   305→
   306→  // nave x pasillo (ESPERA UBICACION)
   307→  const naves = new Map<string, { minutos: number; registros: number; pasillos: Map<string, { minutos: number; registros: number }> }>()
   308→  for (const r of rows) {
   309→    if (r.categoria !== 'ESPERA UBICACION' || !r.nave) continue
   310→    let n = naves.get(r.nave)
   311→    if (!n) { n = { minutos: 0, registros: 0, pasillos: new Map() }; naves.set(r.nave, n) }
   312→    n.minutos += r.minutosEf
   313→    n.registros += 1
   314→    const pk = r.pasillo ?? '?'
   315→    let p = n.pasillos.get(pk)
   316→    if (!p) { p = { minutos: 0, registros: 0 }; n.pasillos.set(pk, p) }
   317→    p.minutos += r.minutosEf
   318→    p.registros += 1
   319→  }
   320→  const porNave = [...naves.entries()].map(([nave, v]) => ({
   321→    nave,
   322→    minutos: v.minutos,
   323→    registros: v.registros,
   324→    pasillos: [...v.pasillos.entries()].map(([pasillo, p]) => ({ pasillo, ...p })).sort((a, b) => b.minutos - a.minutos),
   325→  })).sort((a, b) => b.minutos - a.minutos)
   326→
   327→  // por turno y por dia
   328→  const turnosMap = new Map<string, number>()
   329→  const diasMap = new Map<string, number>()
   330→  const horasMap = new Map<number, number>()
   331→  const detalles = new Map<string, { minutos: number; registros: number }>()
   332→  for (const r of rows) {
   333→    turnosMap.set(r.turno, (turnosMap.get(r.turno) ?? 0) + r.minutosEf)
   334→    const k = dia(r.fecha)
   335→    diasMap.set(k, (diasMap.get(k) ?? 0) + r.minutosEf)
   336→    if (r.horaDesde != null) horasMap.set(Math.floor(r.horaDesde / 60), (horasMap.get(Math.floor(r.horaDesde / 60)) ?? 0) + r.minutosEf)
   337→    const dk = r.detalle ?? '(sin dato)'
   338→    let d = detalles.get(dk)
   339→    if (!d) { d = { minutos: 0, registros: 0 }; detalles.set(dk, d) }
   340→    d.minutos += r.minutosEf
   341→    d.registros += 1
   342→  }
   343→  const NOM_TURNO: Record<string, string> = { TM: 'Mañana', TT: 'Tarde', TN: 'Noche' }
   344→  const porTurno = [...turnosMap.entries()].map(([t, minutos]) => ({ turno: t, nombre: NOM_TURNO[t] ?? t, minutos })).sort((a, b) => b.minutos - a.minutos)
   345→  const porDia = [...diasMap.entries()].sort().map(([fecha, minutos]) => ({ fecha, minutos }))
   346→  const porHora = Array.from({ length: 24 }, (_, h) => ({ hora: h, etiqueta: `${String(h).padStart(2, '0')}:00`, minutos: horasMap.get(h) ?? 0 }))
   347→  const topDetalles = [...detalles.entries()].map(([detalle, v]) => ({ detalle, ...v })).sort((a, b) => b.minutos - a.minutos).slice(0, 25)
   348→
   349→  // cruces codigo x categoria
   350→  const codCat = new Map<string, { code: number | null; categoria: string; minutos: number; registros: number }>()
   351→  for (const r of rows) {
   352→    const k = `${r.motivoCode ?? '-'}|${r.categoria}`
   353→    let c = codCat.get(k)
   354→    if (!c) { c = { code: r.motivoCode, categoria: r.categoria, minutos: 0, registros: 0 }; codCat.set(k, c) }
   355→    c.minutos += r.minutosEf
   356→    c.registros += 1
   357→  }
   358→  const codigoCategoria = [...codCat.values()].sort((a, b) => b.minutos - a.minutos).slice(0, 20)
   359→
   360→  return {
   361→    totalMin,
   362→    totalHoras: +(totalMin / 60).toFixed(1),
   363→    registros: rows.length,
   364→    porCategoria,
   365→    porNave,
   366→    porTurno,
   367→    porDia,
   368→    porHora,
   369→    topDetalles,
   370→    codigoCategoria,
   371→    navesResumen: {
   372→      minutos: porNave.reduce((a, n) => a + n.minutos, 0),
   373→      naves: porNave.length,
   374→      pasillos: porNave.reduce((a, n) => a + n.pasillos.length, 0),
   375→    },
   376→  }
   377→}

   379→// ============ PICKING ============
   380→export async function getPicking(f: Filtros) {
   381→  const where = { fecha: rango(f) }
   382→  const [count, ops] = await Promise.all([
   383→    db.pickingEvento.count({ where }),
   384→    db.pickingEvento.findMany({ where, orderBy: [{ fecha: 'asc' }, { operario: 'asc' }, { horaMin: 'asc' }], select: { fecha: true, operario: true, horaMin: true, bultos: true, soporte: true, circuito: true } }),
   385→  ])
   386→
   387→  if (!count) {
   388→    return { registros: 0, vacio: true as const }
   389→  }
   390→
   391→  // gaps entre eventos consecutivos por operario y dia
   392→  const buckets = [
   393→    { label: '0-1 min', min: 0, max: 1 },
   394→    { label: '1-2 min', min: 1, max: 2 },
   395→    { label: '2-5 min', min: 2, max: 5 },
   396→    { label: '5-10 min', min: 5, max: 10 },
   397→    { label: '10-20 min', min: 10, max: 20 },
   398→    { label: '20-60 min', min: 20, max: 60 },
   399→    { label: '> 60 min', min: 60, max: Infinity },
   400→  ]
   401→  const gapCounts = new Array(buckets.length).fill(0)
   402→  const soporteGapCounts = new Array(buckets.length).fill(0)
   403→  let gapSum = 0, gapN = 0, soporteGapSum = 0, soporteGapN = 0, cambiosSoporte = 0
   404→  const opAgg = new Map<string, { nombre: string; eventos: number; gapSum: number; gapN: number; cambioSop: number; bultos: number }>()
   405→  const serDia = new Map<string, { eventos: number; gapMin: number }>()
   406→
   407→  let prev: { operario: string; key: string; horaMin: number; soporte: string | null } | null = null
   408→  for (const r of ops) {
   409→    const opKey = r.operario
   410→    let o = opAgg.get(opKey)
   411→    if (!o) { o = { nombre: opKey, eventos: 0, gapSum: 0, gapN: 0, cambioSop: 0, bultos: 0 }; opAgg.set(opKey, o) }
   412→    o.eventos += 1
   413→    o.bultos += r.bultos ?? 0
   414→    const k = r.fecha.toISOString().slice(0, 10)
   415→    let s = serDia.get(k)
   416→    if (!s) { s = { eventos: 0, gapMin: 0 }; serDia.set(k, s) }
   417→    s.eventos += 1
   418→
   419→    if (prev && prev.operario === opKey && prev.key === k && r.horaMin != null && prev.horaMin != null) {
   420→      const gap = Math.max(0, r.horaMin - prev.horaMin)
   421→      if (gap > 720) {
   422→        // salto de jornada (>12 h): no es tiempo muerto, se ignora para las metricas
   423→        prev = { operario: opKey, key: k, horaMin: r.horaMin ?? -1, soporte: r.soporte }
   424→        continue
   425→      }
   426→      const bi = buckets.findIndex((b) => gap >= b.min && gap < b.max)
   427→      if (bi >= 0) gapCounts[bi]++
   428→      gapSum += gap; gapN++
   429→      s.gapMin += gap
   430→      o.gapSum += gap; o.gapN++
   431→      const cambioSop = prev.soporte != null && r.soporte != null && prev.soporte !== r.soporte
   432→      if (cambioSop) {
   433→        cambiosSoporte++
   434→        if (bi >= 0) soporteGapCounts[bi]++
   435→        soporteGapSum += gap; soporteGapN++
   436→        o.cambioSop++
   437→      }
   438→    }
   439→    prev = { operario: opKey, key: k, horaMin: r.horaMin ?? -1, soporte: r.soporte }
   440→  }
   441→
   442→  const serie = [...serDia.entries()].sort().map(([fecha, v]) => ({ fecha, eventos: v.eventos, gapPromedio: v.eventos > 1 ? +(v.gapMin / (v.eventos - 1)).toFixed(2) : null }))
   443→
   444→  return {
   445→    registros: count,
   446→    vacio: false as const,
   447→    operarios: opAgg.size,
   448→    gapPromedio: gapN ? +(gapSum / gapN).toFixed(2) : null,
   449→    gapMedianaEst: gapN ? +(gapSum / gapN).toFixed(2) : null,
   450→    cambiosSoporte,
   451→    gapPromedioCambioSoporte: soporteGapN ? +(soporteGapSum / soporteGapN).toFixed(2) : null,
   452→    distribucionGaps: buckets.map((b, i) => ({ bucket: b.label, cantidad: gapCounts[i] })),
   453→    distribucionGapsSoporte: buckets.map((b, i) => ({ bucket: b.label, cantidad: soporteGapCounts[i] })),
   454→    operariosTop: [...opAgg.values()].filter((o) => o.gapN > 30).map((o) => ({ operario: o.nombre, eventos: o.eventos, bultos: o.bultos, gapPromedio: +(o.gapSum / o.gapN).toFixed(2), cambiosSoporte: o.cambioSop })).sort((a, b) => b.eventos - a.eventos).slice(0, 15),
   455→    serie,
   456→  }
   457→}

   459→// ============ PRODUCTIVIDAD POR CIRCUITO/SECTOR (Tiempos E-8 / Productividad X Circuito) ============
   460→const r1 = (x: number) => Math.round(x * 10) / 10
   461→const r2 = (x: number) => Math.round(x * 100) / 100
   462→
   463→export async function getProdCirc(f: Filtros) {
   464→  const where = { fecha: rango(f) }
   465→  const [tot, sectores, filas, ops] = await Promise.all([
   466→    db.prodCircuito.aggregate({
   467→      where,
   468→      _count: true,
   469→      _min: { fecha: true },
   470→      _max: { fecha: true },
   471→      _sum: { soportes: true, lineas: true, bultos: true, tiempoTotal: true, tiempoMuerto: true, tiempoNeto: true, tiempoSuperNeto: true },
   472→    }),
   473→    db.prodCircuito.groupBy({
   474→      by: ['sector'],
   475→      where,
   476→      _sum: { soportes: true, bultos: true, tiempoTotal: true, tiempoMuerto: true, tiempoSuperNeto: true },
   477→    }),
   478→    db.prodCircuito.findMany({ where, select: { fecha: true, bultos: true, tiempoTotal: true, tiempoMuerto: true, tiempoSuperNeto: true } }),
   479→    db.prodCircuito.groupBy({
   480→      by: ['operario', 'nombre'],
   481→      where,
   482→      _sum: { bultos: true, soportes: true, tiempoTotal: true, tiempoMuerto: true, tiempoSuperNeto: true },
   483→    }),
   484→  ])
   485→
   486→  if (!tot._count) return { registros: 0, vacio: true as const }
   487→
   488→  const s = tot._sum
   489→  const kpis = {
   490→    registros: tot._count,
   491→    soportes: Math.round(s.soportes ?? 0),
   492→    lineas: Math.round(s.lineas ?? 0),
   493→    bultos: Math.round(s.bultos ?? 0),
   494→    horasTotal: r1(s.tiempoTotal ?? 0),
   495→    horasMuerto: r1(s.tiempoMuerto ?? 0),
   496→    horasNeto: r1(s.tiempoNeto ?? 0),
   497→    pctMuerto: s.tiempoTotal ? r1(((s.tiempoMuerto ?? 0) / s.tiempoTotal) * 100) : null,
   498→    prodTotal: s.tiempoTotal ? r2((s.bultos ?? 0) / s.tiempoTotal) : null,
   499→    prodNeto: s.tiempoNeto ? r2((s.bultos ?? 0) / s.tiempoNeto) : null,
   500→    prodSuperNeto: s.tiempoSuperNeto ? r2((s.bultos ?? 0) / s.tiempoSuperNeto) : null,
   501→  }
   502→
   503→  const porSector = sectores
   504→    .map((x) => ({
   505→      sector: x.sector || '(sin sector)',
   506→      soportes: Math.round(x._sum.soportes ?? 0),
   507→      bultos: Math.round(x._sum.bultos ?? 0),
   508→      pctMuerto: x._sum.tiempoTotal ? r1(((x._sum.tiempoMuerto ?? 0) / x._sum.tiempoTotal) * 100) : null,
   509→      prodSuperNeto: x._sum.tiempoSuperNeto && x._sum.tiempoSuperNeto > 0 ? r2((x._sum.bultos ?? 0) / x._sum.tiempoSuperNeto) : null,
   510→    }))
   511→    .sort((a, b) => b.bultos - a.bultos)
   512→
   513→  // serie mensual
   514→  const porMesMap = new Map<string, { bultos: number; tiempoTotal: number; tiempoMuerto: number; tiempoSuperNeto: number }>()
   515→  for (const r of filas) {
   516→    const k = dia(r.fecha).slice(0, 7)
   517→    let m = porMesMap.get(k)
   518→    if (!m) { m = { bultos: 0, tiempoTotal: 0, tiempoMuerto: 0, tiempoSuperNeto: 0 }; porMesMap.set(k, m) }
   519→    m.bultos += r.bultos ?? 0
   520→    m.tiempoTotal += r.tiempoTotal ?? 0
   521→    m.tiempoMuerto += r.tiempoMuerto ?? 0
   522→    m.tiempoSuperNeto += r.tiempoSuperNeto ?? 0
   523→  }
   524→  const porMes = [...porMesMap.entries()].sort(([a], [b]) => a.localeCompare(b)).map(([mes, m]) => ({
   525→    mes,
   526→    bultos: Math.round(m.bultos),
   527→    pctMuerto: m.tiempoTotal ? r1((m.tiempoMuerto / m.tiempoTotal) * 100) : null,
   528→    prodSuperNeto: m.tiempoSuperNeto > 0 ? r2(m.bultos / m.tiempoSuperNeto) : null,
   529→  }))
   530→
   531→  const operariosTop = ops
   532→    .map((x) => ({
   533→      operario: x.operario,
   534→      nombre: x.nombre || x.operario,
   535→      bultos: Math.round(x._sum.bultos ?? 0),
   536→      soportes: Math.round(x._sum.soportes ?? 0),
   537→      horasTotal: r1(x._sum.tiempoTotal ?? 0),
   538→      pctMuerto: x._sum.tiempoTotal ? r1(((x._sum.tiempoMuerto ?? 0) / x._sum.tiempoTotal) * 100) : null,
   539→      prodSuperNeto: x._sum.tiempoSuperNeto && x._sum.tiempoSuperNeto > 0 ? r2((x._sum.bultos ?? 0) / x._sum.tiempoSuperNeto) : null,
   540→    }))
   541→    .sort((a, b) => b.bultos - a.bultos)
   542→    .slice(0, 15)
   543→
   544→  return {
   545→    registros: tot._count,
   546→    vacio: false as const,
   547→    desde: tot._min.fecha ? dia(tot._min.fecha) : undefined,
   548→    hasta: tot._max.fecha ? dia(tot._max.fecha) : undefined,
   549→    kpis,
   550→    porSector,
   551→    porMes,
   552→    operariosTop,
   553→  }
   554→}
