    12→function rango(f: Filtros): { gte?: Date; lte?: Date } {
    13→  const r: { gte?: Date; lte?: Date } = {}
    14→  if (f.from) r.gte = new Date(f.from + 'T00:00:00.000Z')
    15→  if (f.to) r.lte = new Date(f.to + 'T23:59:59.999Z')
    16→  return r
    17→}
    18→const dia = (d: Date) => d.toISOString().slice(0, 10)

    20→// ============ RESUMEN EJECUTIVO ============
    21→export async function getResumen() {
    22→  const [olaCount, h61Count, tmCount, pickCount, batches] = await Promise.all([
    23→    db.olaDia.aggregate({ _count: true, _min: { fecha: true }, _max: { fecha: true }, _avg: { ola: true, total: true } }),
    24→    db.h61OpDia.aggregate({ _count: true, _min: { fecha: true }, _max: { fecha: true }, _sum: { bultos: true, horasActivas: true, extras: true, bultosExtras: true } }),
    25→    db.tiempoMuerto.aggregate({ _count: true, _min: { fecha: true }, _max: { fecha: true }, _sum: { minutos: true, minutosEf: true } }),
    26→    db.pickingEvento.aggregate({ _count: true, _min: { fecha: true }, _max: { fecha: true } }),
    27→    db.uploadBatch.findMany({ orderBy: { createdAt: 'desc' }, take: 10 }),
    28→  ])
    29→
    30→  const opsUnicos = await db.h61OpDia.findMany({ select: { operario: true }, distinct: ['operario'] })
    31→
    32→  // top categorias TM (excluye bajas ESTADO B; OR con null es clave: NOT excludes NULLs en SQL)
    33→  const tmCats = await db.tiempoMuerto.groupBy({
    34→    by: ['categoria'],
    35→    _sum: { minutosEf: true },
    36→    _count: true,
    37→    where: { OR: [{ estado: null }, { estado: { not: 'B' } }] },
    38→    orderBy: { _sum: { minutosEf: 'desc' } },
    39→  })
    40→
    41→  const horasTotales = h61Count._sum.horasActivas ?? 0
    42→  const bultos = h61Count._sum.bultos ?? 0
    43→  const bultosExtras = h61Count._sum.bultosExtras ?? 0
    44→  const totalMinTM = tmCats.reduce((a, c) => a + (c._sum.minutosEf ?? 0), 0)
    45→
    46→  return {
    47→    datasets: {
    48→      ola: { registros: olaCount._count, desde: olaCount._min.fecha, hasta: olaCount._max.fecha, olaMedia: Math.round(olaCount._avg.ola ?? 0), totalMedia: Math.round(olaCount._avg.total ?? 0) },
    49→      h61: { registros: h61Count._count, desde: h61Count._min.fecha, hasta: h61Count._max.fecha, bultos, horas: horasTotales, productividad: horasTotales ? +(bultos / horasTotales).toFixed(1) : 0, operarios: opsUnicos.length, extrasHoras: h61Count._sum.extras ?? 0, bultosExtras, pctBultosExtras: bultos ? +((bultosExtras / bultos) * 100).toFixed(1) : 0 },
    50→      tm: { registros: tmCount._count, desde: tmCount._min.fecha, hasta: tmCount._max.fecha, minutos: totalMinTM, horas: +(totalMinTM / 60).toFixed(0), pctSobreHorasH61: horasTotales ? +(((totalMinTM / 60) / horasTotales) * 100).toFixed(1) : 0 },
    51→      picking: { registros: pickCount._count, desde: pickCount._min.fecha, hasta: pickCount._max.fecha },
    52→    },
    53→    topCategoriasTM: tmCats.slice(0, 6).map((c) => ({ categoria: c.categoria, minutos: c._sum.minutosEf ?? 0, registros: c._count })),
    54→    batches: batches.map((b) => ({ ...b, createdAt: b.createdAt.toISOString() })),
    55→  }
    56→}

    58→// ============ OLA / PLANIFICACION ============
    59→export async function getPlanificacion(f: Filtros) {
    60→  const where = { fecha: rango(f) }
    61→  const [ola, h61] = await Promise.all([
    62→    db.olaDia.findMany({ where, orderBy: { fecha: 'asc' } }),
    63→    db.h61OpDia.findMany({ where: { fecha: rango(f) }, select: { fecha: true, operario: true, horasActivas: true, bultos: true, bultosBase: true, bultosExtras: true, extras: true, turno: true, funcion: true } }),
    64→  ])
    65→
    66→  const porDia = new Map<string, { ops: number; opsExtras: number; horas: number; horasExtras: number; bultos: number; bultosBase: number; bultosExtras: number }>()
    67→  for (const r of h61) {
    68→    const k = dia(r.fecha)
    69→    let p = porDia.get(k)
    70→    if (!p) { p = { ops: 0, opsExtras: 0, horas: 0, horasExtras: 0, bultos: 0, bultosBase: 0, bultosExtras: 0 }; porDia.set(k, p) }
    71→    p.ops += 1
    72→    if (r.extras > 0) p.opsExtras += 1
    73→    p.horas += r.horasActivas
    74→    p.horasExtras += r.extras
    75→    p.bultos += r.bultos
    76→    p.bultosBase += r.bultosBase
    77→    p.bultosExtras += r.bultosExtras
    78→  }
    79→
    80→  const fechas = new Set<string>([...ola.map((o) => dia(o.fecha)), ...porDia.keys()])
    81→  const serie = [...fechas].sort().map((k) => {
    82→    const o = ola.find((x) => dia(x.fecha) === k)
    83→    const p = porDia.get(k)
    84→    const demanda = o?.total ?? null
    85→    const prodHora = p && p.horas ? p.bultos / p.horas : null
    86→    // capacidad si se suprimen extras: mantiene bultosBase (produccion en <=8h por operario)
    87→    const faltanteSinExtras = p && demanda != null ? Math.max(0, demanda - p.bultosBase) : null
    88→    return {
    89→      fecha: k,
    90→      ola: o?.ola ?? null,
    91→      pendiente: o?.pendiente ?? null,
    92→      demanda,
    93→      preparado: p?.bultos ?? null,
    94→      ops: p?.ops ?? null,
    95→      opsExtras: p?.opsExtras ?? null,
    96→      horas: p ? +p.horas.toFixed(1) : null,
    97→      horasExtras: p?.horasExtras ?? 0,
    98→      prodHora: prodHora != null ? +prodHora.toFixed(1) : null,
    99→      bultosBase: p?.bultosBase ?? null,
   100→      bultosExtras: p?.bultosExtras ?? null,
   101→      faltanteSinExtras: faltanteSinExtras != null ? Math.round(faltanteSinExtras) : null,
   102→    }
   103→  })
   104→
   105→  // agregados por mes
   106→  const porMes = new Map<string, { ola: number; demanda: number; preparado: number; dias: number; diasExtras: number; horasExtras: number; faltante: number }>()
   107→  for (const s of serie) {
   108→    const k = s.fecha.slice(0, 7)
   109→    let m = porMes.get(k)
   110→    if (!m) { m = { ola: 0, demanda: 0, preparado: 0, dias: 0, diasExtras: 0, horasExtras: 0, faltante: 0 }; porMes.set(k, m) }
   111→    m.dias += 1
   112→    if (s.demanda != null) m.demanda += s.demanda
   113→    if (s.ola != null) m.ola += s.ola
   114→    if (s.preparado != null) m.preparado += s.preparado
   115→    if ((s.opsExtras ?? 0) > 0) m.diasExtras += 1
   116→    m.horasExtras += s.horasExtras ?? 0
   117→    if (s.faltanteSinExtras != null) m.faltante += s.faltanteSinExtras
   118→  }
   119→  const meses = [...porMes.entries()].sort().map(([mes, v]) => ({ mes, ...v }))
   120→
   121→  const conExtras = serie.filter((s) => (s.opsExtras ?? 0) > 0)
   122→  const resumen = {
   123→    dias: serie.length,
   124→    diasConExtras: conExtras.length,
   125→    pctDiasConExtras: serie.length ? +((conExtras.length / serie.length) * 100).toFixed(1) : 0,
   126→    faltantePromedioSinExtras: conExtras.length ? Math.round(conExtras.reduce((a, s) => a + (s.faltanteSinExtras ?? 0), 0) / conExtras.length) : 0,
   127→    faltanteTotalSinExtras: conExtras.reduce((a, s) => a + (s.faltanteSinExtras ?? 0), 0),
   128→    horasExtrasTotales: serie.reduce((a, s) => a + (s.horasExtras ?? 0), 0),
   129→    bultosExtrasTotales: serie.reduce((a, s) => a + (s.bultosExtras ?? 0), 0),
   130→    prodHoraMedia: (() => {
   131→      const vals = serie.filter((s) => s.prodHora != null && s.preparado != null && (s.preparado ?? 0) > 0)
   132→      const totB = vals.reduce((a, s) => a + (s.preparado ?? 0), 0)
   133→      const totH = vals.reduce((a, s) => a + (s.horas ?? 0), 0)
   134→      return totH ? +(totB / totH).toFixed(1) : null
   135→    })(),
   136→  }
   137→
   138→  return { serie, meses, resumen }
   139→}

   141→// ============ OLA (analisis exclusivo de ola y pendientes) ============
   142→// Serie diaria cruda + lunes de cada semana ISO. Los agregados (mes/semana/dia de
   143→// semana, promedio y mediana) se recalculan en el cliente para que los filtros
   144→// mensual/semanal/diario sean consistentes entre si.
   145→const DIAS_SEM = ['Domingo', 'Lunes', 'Martes', 'Miércoles', 'Jueves', 'Viernes', 'Sábado']
   146→
   147→export async function getOla(f: Filtros) {
   148→  const rows = await db.olaDia.findMany({ where: { fecha: rango(f) }, orderBy: { fecha: 'asc' } })
   149→  const serie = rows.map((r) => {
   150→    const k = dia(r.fecha)
   151→    const d = new Date(k + 'T00:00:00.000Z')
   152→    const dow = d.getUTCDay() // 0=domingo
   153→    const lun = new Date(d)
   154→    lun.setUTCDate(lun.getUTCDate() - ((dow + 6) % 7)) // lunes de la semana
   155→    const ola = r.ola ?? 0
   156→    const pendiente = r.pendiente ?? 0
   157→    return {
   158→      fecha: k,
   159→      diaSemana: DIAS_SEM[dow],
   160→      esFinde: dow === 0 || dow === 6,
   161→      semana: lun.toISOString().slice(0, 10),
   162→      ola,
   163→      pendiente,
   164→      total: ola + pendiente,
   165→    }
   166→  })
   167→  return { serie }
   168→}

   170→// ============ PRODUCTIVIDAD H61 ============
   171→export async function getH61(f: Filtros) {
   172→  const where: Record<string, unknown> = { fecha: rango(f) }
   173→  if (f.turno) where.turno = f.turno
   174→  if (f.funcion) where.funcion = f.funcion
   175→
   176→  const [ops, th, ci] = await Promise.all([
   177→    db.h61OpDia.findMany({ where }),
   178→    db.h61TurnoHora.findMany({ where: { fecha: rango(f) } }),
   179→    db.h61Circuito.findMany({ where: { fecha: rango(f) } }),
   180→  ])
   181→
   182→  // Perfil 24h: horas 0-5 del turno noche (N) pertenecen al dia calendario siguiente
   183→  const porHora = Array.from({ length: 24 }, () => ({ bultos: 0, operarios: new Set<string>() }))
   184→  const opSeen = new Map<string, Set<number>>() // fecha|op -> horas
   185→  for (const r of th) {
   186→    const fechaBase = new Date(r.fecha)
   187→    const esNocheMadrugada = r.turno === 'N' && r.hora <= 5
   188→    if (esNocheMadrugada) fechaBase.setUTCDate(fechaBase.getUTCDate() + 1)
   189→    const h = porHora[r.hora]
   190→    h.bultos += r.bultos
   191→    const k = `${dia(fechaBase)}|${r.turno}`
   192→    // operarios por hora aproximamos con el maximo informado ese dia-turno
   193→    const set = opSeen.get(k) ?? new Set<number>()
   194→    set.add(r.operarios)
   195→    opSeen.set(k, set)
   196→  }
   197→  const perfilHora = porHora.map((h, i) => ({
   198→    hora: i,
   199→    etiqueta: `${String(i).padStart(2, '0')}:00`,
   200→    bultos: h.bultos,
   201→    operariosProm: (() => {
   202→      const vals = [...opSeen.values()].map((s) => Math.max(...s))
   203→      return vals.length ? +(vals.reduce((a, b) => a + b, 0) / vals.length).toFixed(1) : 0
   204→    })(),
   205→  }))
   206→  const horasConDatos = perfilHora.filter((p) => p.bultos > 0)
   207→  const pico = horasConDatos.reduce((a, b) => (b.bultos > a.bultos ? b : a), horasConDatos[0] ?? { hora: 0, etiqueta: '-', bultos: 0, operariosProm: 0 })
   208→  const valle = horasConDatos.reduce((a, b) => (b.bultos < a.bultos ? b : a), horasConDatos[0] ?? { hora: 0, etiqueta: '-', bultos: 0, operariosProm: 0 })
   209→
   210→  // por turno
   211→  const turnoAgg = new Map<string, { bultos: number; horas: number; opsDias: number; bultosExtras: number }>()
   212→  for (const r of th) {
   213→    let t = turnoAgg.get(r.turno)
   214→    if (!t) { t = { bultos: 0, horas: 0, opsDias: 0, bultosExtras: 0 }; turnoAgg.set(r.turno, t) }
   215→    t.bultos += r.bultos
   216→  }
   217→  const turnoHoras = new Map<string, Set<string>>()
   218→  for (const r of ops) {
   219→    let t = turnoHoras.get(r.turno)
   220→    if (!t) { t = new Set(); turnoHoras.set(r.turno, t) }
   221→    t.add(`${dia(r.fecha)}|${r.operario}`)
   222→    t.add(`${dia(r.fecha)}|${r.operario}|h|${r.horasActivas}`)
   223→  }
   224→  for (const [turno, set] of turnoHoras) {
   225→    const horas = [...set].filter((x) => x.includes('|h|')).reduce((a, x) => a + parseInt(x.split('|h|')[1], 10), 0)
   226→    const opsDias = [...set].filter((x) => !x.includes('|h|')).length
   227→    const t = turnoAgg.get(turno)
   228→    if (t) { t.horas = horas; t.opsDias = opsDias }
   229→  }
   230→  for (const r of ops) {
   231→    const t = turnoAgg.get(r.turno)
   232→    if (t) t.bultosExtras += r.bultosExtras
   233→  }
   234→  const porTurno = [...turnoAgg.entries()].map(([turno, v]) => ({
   235→    turno,
   236→    nombre: turno === 'M' ? 'Mañana' : turno === 'T' ? 'Tarde' : turno === 'N' ? 'Noche' : turno,
   237→    ...v,
   238→    prodHora: v.horas ? +(v.bultos / v.horas).toFixed(1) : 0,
   239→  })).sort((a, b) => b.bultos - a.bultos)
   240→
   241→  // heatmap turno x hora
   242→  const heat = ['M', 'T', 'N'].map((turno) => {
   243→    const fila = Array.from({ length: 24 }, () => 0)
   244→    for (const r of th) if (r.turno === turno) fila[r.hora] = Math.max(fila[r.hora], Math.round(r.bultos / Math.max(1, r.operarios)))
   245→    return { turno, valores: fila }
   246→  })
   247→
   248→  // circuitos
   249→  const ciAgg = new Map<string, number>()
   250→  for (const r of ci) ciAgg.set(r.circuito, (ciAgg.get(r.circuito) ?? 0) + r.bultos)
   251→  const circuitos = [...ciAgg.entries()].map(([circuito, bultos]) => ({ circuito, bultos })).sort((a, b) => b.bultos - a.bultos)
   252→
   253→  // distribucion de jornadas 8h vs extras
   254→  const distHoras = new Map<number, number>()
   255→  for (const r of ops) distHoras.set(r.horasActivas, (distHoras.get(r.horasActivas) ?? 0) + 1)
   256→  const distribucion = [...distHoras.entries()].sort((a, b) => a[0] - b[0]).map(([horas, opsDia]) => ({ horas, opsDia }))
   257→
   258→  const opsExtras = ops.filter((r) => r.extras > 0)
   259→  const extrasResumen = {
   260→    opsDiasTotal: ops.length,
   261→    opsDiasConExtras: opsExtras.length,
   262→    pctOpsDiasConExtras: ops.length ? +((opsExtras.length / ops.length) * 100).toFixed(1) : 0,
   263→    horasExtras: opsExtras.reduce((a, r) => a + r.extras, 0),
   264→    bultosExtras: opsExtras.reduce((a, r) => a + r.bultosExtras, 0),
   265→    bultosTotales: ops.reduce((a, r) => a + r.bultos, 0),
   266→  }
   267→  extrasResumen.pctBultosExtras = extrasResumen.bultosTotales ? +((extrasResumen.bultosExtras / extrasResumen.bultosTotales) * 100).toFixed(1) : 0
   268→
   269→  // top operarios por productividad (minimo 80 horas)
   270→  const opAgg = new Map<string, { nombre: string | null; bultos: number; horas: number; extras: number; dias: number; funcion: string }>()
   271→  for (const r of ops) {
   272→    let o = opAgg.get(r.operario)
   273→    if (!o) { o = { nombre: r.nombre, bultos: 0, horas: 0, extras: 0, dias: 0, funcion: r.funcion }; opAgg.set(r.operario, o) }
   274→    if (!o.nombre && r.nombre) o.nombre = r.nombre
   275→    o.bultos += r.bultos
   276→    o.horas += r.horasActivas
   277→    o.extras += r.extras
   278→    o.dias += 1
   279→  }
   280→  const operarios = [...opAgg.entries()]
   281→    .map(([operario, o]) => ({ operario, nombre: o.nombre ?? operario, bultos: o.bultos, horas: o.horas, extras: o.extras, dias: o.dias, prodHora: o.horas ? +(o.bultos / o.horas).toFixed(1) : 0 }))
   282→    .filter((o) => o.horas >= 80)
   283→    .sort((a, b) => b.prodHora - a.prodHora)
   284→
   285→  // serie diaria
   286→  const serDia = new Map<string, { bultos: number; horas: number; ops: number }>()
   287→  for (const r of ops) {
   288→    const k = dia(r.fecha)
   289→    let s = serDia.get(k)
   290→    if (!s) { s = { bultos: 0, horas: 0, ops: 0 }; serDia.set(k, s) }
   291→    s.bultos += r.bultos
   292→    s.horas += r.horasActivas
   293→    s.ops += 1
   294→  }
   295→  const serie = [...serDia.entries()].sort().map(([fecha, v]) => ({ fecha, ...v, prodHora: v.horas ? +(v.bultos / v.horas).toFixed(1) : 0 }))
   296→
   297→  const bultosTot = ops.reduce((a, r) => a + r.bultos, 0)
   298→  const horasTot = ops.reduce((a, r) => a + r.horasActivas, 0)
   299→
   300→  return {
   301→    perfilHora,
   302→    pico: { ...pico },
   303→    valle: { ...valle },
   304→    porTurno,
   305→    heat,
   306→    circuitos: circuitos.slice(0, 17),
   307→    distribucion,
   308→    extrasResumen,
   309→    operariosTop: operarios.slice(0, 15),
   310→    operariosBottom: operarios.slice(-15).reverse(),
   311→    operariosExtras: [...opAgg.entries()].map(([operario, o]) => ({ operario, nombre: o.nombre ?? operario, extras: o.extras, dias: o.dias })).filter((o) => o.extras > 0).sort((a, b) => b.extras - a.extras).slice(0, 15),
   312→    serie,
   313→    resumen: { bultos: bultosTot, horas: horasTot, prodHora: horasTot ? +(bultosTot / horasTot).toFixed(1) : 0, opsUnicos: opAgg.size },
   314→  }
   315→}

   317→// ============ CAPACIDAD H61 (ritmo de preparacion: jornada vs extras) ============
   318→// Regla: las primeras 8 horas con produccion de cada operario-dia cuentan como
   319→// jornada base; el excedente de horas y sus bultos cuentan como horas extra.
   320→function mediana(vals: number[]): number {
   321→  if (!vals.length) return 0
   322→  const s = [...vals].sort((a, b) => a - b)
   323→  const mid = Math.floor(s.length / 2)
   324→  return s.length % 2 ? s[mid] : (s[mid - 1] + s[mid]) / 2
   325→}

   327→export async function getCapacidad(f: Filtros) {
   328→  const [ops, oh] = await Promise.all([
   329→    db.h61OpDia.findMany({ where: { fecha: rango(f) }, orderBy: { fecha: 'asc' } }),
   330→    db.h61OpHora.findMany({ where: { fecha: rango(f) } }),
   331→  ])
   332→  if (!ops.length) return { serie: [], porMes: [], porTurno: [], perfilHora: [], resumen: null, tieneOpHora: oh.length > 0 }
   333→
   334→  // --- serie diaria ---
   335→  const porDia = new Map<string, { bultos: number; bultosBase: number; bultosExtras: number; horas: number; horasExtras: number; ops: Set<string>; opsExtras: Set<string>; opDias: number }>()
   336→  for (const r of ops) {
   337→    const k = dia(r.fecha)
   338→    let p = porDia.get(k)
   339→    if (!p) { p = { bultos: 0, bultosBase: 0, bultosExtras: 0, horas: 0, horasExtras: 0, ops: new Set(), opsExtras: new Set(), opDias: 0 }; porDia.set(k, p) }
   340→    p.bultos += r.bultos
   341→    p.bultosBase += r.bultosBase
   342→    p.bultosExtras += r.bultosExtras
   343→    p.horas += r.horasActivas
   344→    p.horasExtras += r.extras
   345→    p.ops.add(r.operario)
   346→    if (r.extras > 0) p.opsExtras.add(r.operario)
   347→    p.opDias += 1
   348→  }
   349→
   350→  const serie = [...porDia.entries()].sort().map(([fecha, p]) => ({
   351→    fecha,
   352→    bultos: p.bultos,
   353→    bultosBase: p.bultosBase,
   354→    bultosExtras: p.bultosExtras,
   355→    pctExtras: p.bultos ? +((p.bultosExtras / p.bultos) * 100).toFixed(1) : 0,
   356→    horas: p.horas,
   357→    horasExtras: p.horasExtras,
   358→    ritmo: p.horas ? +(p.bultos / p.horas).toFixed(1) : null,
   359→    ops: p.ops.size,
   360→    opsExtras: p.opsExtras.size,
   361→    opDias: p.opDias,
   362→  }))
   363→
   364→  // --- agregados por mes ---
   365→  const mesAgg = new Map<string, { bultos: number; bultosBase: number; bultosExtras: number; horas: number; horasExtras: number; opDias: number; ops: Set<string>; opsExtras: Set<string>; ritmos: number[]; dias: number }>()
   366→  for (const s of serie) {

   370→    m.dias += 1
   371→    m.bultos += s.bultos
   372→    m.bultosBase += s.bultosBase
   373→    m.bultosExtras += s.bultosExtras
   374→    m.horas += s.horas
   375→    m.horasExtras += s.horasExtras
   376→    m.opDias += s.opDias
   377→    m.ritmos.push(s.ritmo ?? 0)
   378→  }
   379→  // personas distintas por mes (necesita operarios por dia)
   380→  const opsPorDia = new Map<string, { ops: Set<string>; opsExtras: Set<string> }>()
   381→  for (const r of ops) {
   382→    const k = dia(r.fecha)
   383→    let p = opsPorDia.get(k)
   384→    if (!p) { p = { ops: new Set(), opsExtras: new Set() }; opsPorDia.set(k, p) }
   385→    p.ops.add(r.operario)
   386→    if (r.extras > 0) p.opsExtras.add(r.operario)
   387→  }
   388→  for (const s of serie) {
   389→    const m = mesAgg.get(s.fecha.slice(0, 7))
   390→    const p = opsPorDia.get(s.fecha)
   391→    if (!m || !p) continue
   392→    for (const op of p.ops) m.ops.add(op)
   393→    for (const op of p.opsExtras) m.opsExtras.add(op)
   394→  }
   395→  const porMes = [...mesAgg.entries()].sort().map(([mes, m]) => ({
   396→    mes,
   397→    dias: m.dias,
   398→    actividades: m.opDias,
   399→    personas: m.ops.size,
   400→    personasExtras: m.opsExtras.size,

   403→    bultosExtras: m.bultosExtras,

   405→    horas: m.horas,

   408→    ritmoMediana: m.ritmos.length ? +mediana(m.ritmos).toFixed(1) : null,
   409→  }))
   410→
   411→  // --- perfil por hora: colaboradores en jornada vs extras (promedio por dia) ---
   412→  type HoraAgg = { fechas: Set<string>; jornada: Map<string, Set<string>>; extras: Map<string, Set<string>>; bultos: Map<string, number> }
   413→  const horasAgg: HoraAgg[] = Array.from({ length: 24 }, () => ({ fechas: new Set<string>(), jornada: new Map(), extras: new Map(), bultos: new Map() }))

   423→    h.bultos.set(k, (h.bultos.get(k) ?? 0) + r.bultos)
   424→  }
   425→  const perfilHora = horasAgg.map((h, hora) => {
   426→    const n = h.fechas.size
   427→    const sumaJornada = [...h.jornada.values()].reduce((a, s) => a + s.size, 0)
   428→    const sumaExtras = [...h.extras.values()].reduce((a, s) => a + s.size, 0)
   429→    const sumaBultos = [...h.bultos.values()].reduce((a, b) => a + b, 0)
   430→    return {
   431→      hora,
   432→      etiqueta: `${String(hora).padStart(2, '0')}:00`,
   433→      opsJornada: n ? +(sumaJornada / n).toFixed(1) : 0,
   434→      opsExtras: n ? +(sumaExtras / n).toFixed(1) : 0,
   435→      bultosProm: n ? Math.round(sumaBultos / n) : 0,
   436→    }
   437→  })
   438→
   439→  // --- resumen por turno (ventanas: TM 6-14, TT 14-22, TN 23-06) ---
   440→  const turnoAgg = new Map<string, { bultos: number; bultosBase: number; bultosExtras: number; horas: number; horasExtras: number; opDias: number; ops: Set<string>; opsExtras: Set<string> }>()
   441→  for (const r of ops) {
   442→    let t = turnoAgg.get(r.turno)
   443→    if (!t) { t = { bultos: 0, bultosBase: 0, bultosExtras: 0, horas: 0, horasExtras: 0, opDias: 0, ops: new Set(), opsExtras: new Set() }; turnoAgg.set(r.turno, t) }
   444→    t.bultos += r.bultos
   445→    t.bultosBase += r.bultosBase
   446→    t.bultosExtras += r.bultosExtras
   447→    t.horas += r.horasActivas
   448→    t.horasExtras += r.extras
   449→    t.opDias += 1
   450→    t.ops.add(r.operario)
   451→    if (r.extras > 0) t.opsExtras.add(r.operario)
   452→  }
   453→  const NOMBRE_TURNO: Record<string, string> = { M: 'TM (6 a 14)', T: 'TT (14 a 22)', N: 'TN (23 a 06)' }
   454→  const porTurno = [...turnoAgg.entries()].sort().map(([turno, t]) => ({
   455→    turno,
   456→    nombre: NOMBRE_TURNO[turno] ?? turno,
   457→    opDias: t.opDias,
   458→    personas: t.ops.size,
   459→    personasExtras: t.opsExtras.size,
   460→    bultos: t.bultos,
   461→    bultosBase: t.bultosBase,
   462→    bultosExtras: t.bultosExtras,
   463→    pctExtras: t.bultos ? +((t.bultosExtras / t.bultos) * 100).toFixed(1) : 0,
   464→    horasExtras: t.horasExtras,
   465→    ritmoProm: t.horas ? +(t.bultos / t.horas).toFixed(1) : null,
   466→  }))
   467→
   468→  // --- resumen del periodo filtrado ---
   469→  const bultos = serie.reduce((a, s) => a + s.bultos, 0)
   470→  const bultosBase = serie.reduce((a, s) => a + s.bultosBase, 0)
   471→  const bultosExtras = serie.reduce((a, s) => a + s.bultosExtras, 0)
   472→  const horas = serie.reduce((a, s) => a + s.horas, 0)
   473→  const horasExtras = serie.reduce((a, s) => a + s.horasExtras, 0)
   474→  const personas = new Set<string>()
   475→  const personasExtras = new Set<string>()
   476→  for (const p of opsPorDia.values()) {
   477→    for (const op of p.ops) personas.add(op)
   478→    for (const op of p.opsExtras) personasExtras.add(op)
   479→  }
   480→  const resumen = {
   481→    dias: serie.length,
   482→    actividades: serie.reduce((a, s) => a + s.opDias, 0),
   483→    personas: personas.size,
   484→    personasExtras: personasExtras.size,
   485→    bultos,
   486→    bultosBase,
   487→    bultosExtras,
   488→    pctExtras: bultos ? +((bultosExtras / bultos) * 100).toFixed(1) : 0,
   489→    horas,
   490→    horasExtras,
   491→    ritmoProm: horas ? +(bultos / horas).toFixed(1) : null,
   492→    ritmoMediana: +mediana(serie.map((s) => s.ritmo ?? 0)).toFixed(1),
   493→  }
   494→
   495→  return { serie, porMes, porTurno, perfilHora, resumen, tieneOpHora: oh.length > 0 }
   496→}

   499→export async function getTM(f: Filtros & { incluirBajas?: boolean }) {
   500→  const where: Record<string, unknown> = { fecha: rango(f) }
   501→  if (!f.incluirBajas) where.OR = [{ estado: null }, { estado: { not: 'B' } }]
   502→  if (f.turno) where.turno = f.turno
   503→
   504→  const rows = await db.tiempoMuerto.findMany({ where })
   505→
   506→  const totalMin = rows.reduce((a, r) => a + r.minutosEf, 0)
   507→  const cats = new Map<string, { minutos: number; registros: number }>()
   508→  for (const r of rows) {
   509→    let c = cats.get(r.categoria)
   510→    if (!c) { c = { minutos: 0, registros: 0 }; cats.set(r.categoria, c) }
   511→    c.minutos += r.minutosEf
   512→    c.registros += 1
   513→  }
   514→  const porCategoria = [...cats.entries()].map(([categoria, v]) => ({ categoria, ...v, pct: totalMin ? +((v.minutos / totalMin) * 100).toFixed(1) : 0 })).sort((a, b) => b.minutos - a.minutos)
   515→
   516→  // nave x pasillo (ESPERA UBICACION)
   517→  const naves = new Map<string, { minutos: number; registros: number; pasillos: Map<string, { minutos: number; registros: number }> }>()
   518→  for (const r of rows) {
   519→    if (r.categoria !== 'ESPERA UBICACION' || !r.nave) continue
   520→    let n = naves.get(r.nave)
   521→    if (!n) { n = { minutos: 0, registros: 0, pasillos: new Map() }; naves.set(r.nave, n) }
   522→    n.minutos += r.minutosEf
   523→    n.registros += 1
   524→    const pk = r.pasillo ?? '?'
   525→    let p = n.pasillos.get(pk)
   526→    if (!p) { p = { minutos: 0, registros: 0 }; n.pasillos.set(pk, p) }
   527→    p.minutos += r.minutosEf
   528→    p.registros += 1
   529→  }
   530→  const porNave = [...naves.entries()].map(([nave, v]) => ({
   531→    nave,
   532→    minutos: v.minutos,
   533→    registros: v.registros,
   534→    pasillos: [...v.pasillos.entries()].map(([pasillo, p]) => ({ pasillo, ...p })).sort((a, b) => b.minutos - a.minutos),
   535→  })).sort((a, b) => b.minutos - a.minutos)
   536→
   537→  // por turno y por dia
   538→  const turnosMap = new Map<string, number>()
   539→  const diasMap = new Map<string, number>()
   540→  const horasMap = new Map<number, number>()
   541→  const detalles = new Map<string, { minutos: number; registros: number }>()
   542→  for (const r of rows) {
   543→    turnosMap.set(r.turno, (turnosMap.get(r.turno) ?? 0) + r.minutosEf)
   544→    const k = dia(r.fecha)
   545→    diasMap.set(k, (diasMap.get(k) ?? 0) + r.minutosEf)
   546→    if (r.horaDesde != null) horasMap.set(Math.floor(r.horaDesde / 60), (horasMap.get(Math.floor(r.horaDesde / 60)) ?? 0) + r.minutosEf)
   547→    const dk = r.detalle ?? '(sin dato)'
   548→    let d = detalles.get(dk)
   549→    if (!d) { d = { minutos: 0, registros: 0 }; detalles.set(dk, d) }
   550→    d.minutos += r.minutosEf
   551→    d.registros += 1
   552→  }
   553→  const NOM_TURNO: Record<string, string> = { TM: 'Mañana', TT: 'Tarde', TN: 'Noche' }
   554→  const porTurno = [...turnosMap.entries()].map(([t, minutos]) => ({ turno: t, nombre: NOM_TURNO[t] ?? t, minutos })).sort((a, b) => b.minutos - a.minutos)
   555→  const porDia = [...diasMap.entries()].sort().map(([fecha, minutos]) => ({ fecha, minutos }))
   556→  const porHora = Array.from({ length: 24 }, (_, h) => ({ hora: h, etiqueta: `${String(h).padStart(2, '0')}:00`, minutos: horasMap.get(h) ?? 0 }))
   557→  const topDetalles = [...detalles.entries()].map(([detalle, v]) => ({ detalle, ...v })).sort((a, b) => b.minutos - a.minutos).slice(0, 25)
   558→
   559→  // cruces codigo x categoria
   560→  const codCat = new Map<string, { code: number | null; categoria: string; minutos: number; registros: number }>()
   561→  for (const r of rows) {
   562→    const k = `${r.motivoCode ?? '-'}|${r.categoria}`
   563→    let c = codCat.get(k)
   564→    if (!c) { c = { code: r.motivoCode, categoria: r.categoria, minutos: 0, registros: 0 }; codCat.set(k, c) }
   565→    c.minutos += r.minutosEf
   566→    c.registros += 1
   567→  }
   568→  const codigoCategoria = [...codCat.values()].sort((a, b) => b.minutos - a.minutos).slice(0, 20)
   569→
   570→  return {
   571→    totalMin,
   572→    totalHoras: +(totalMin / 60).toFixed(1),
   573→    registros: rows.length,
   574→    porCategoria,
   575→    porNave,
   576→    porTurno,
   577→    porDia,
   578→    porHora,
   579→    topDetalles,
   580→    codigoCategoria,
   581→    navesResumen: {
   582→      minutos: porNave.reduce((a, n) => a + n.minutos, 0),
   583→      naves: porNave.length,
   584→      pasillos: porNave.reduce((a, n) => a + n.pasillos.length, 0),
   585→    },
   586→  }
   587→}

   589→// ============ PICKING ============
   590→export async function getPicking(f: Filtros) {
   591→  const where = { fecha: rango(f) }
   592→  const [count, ops] = await Promise.all([
   593→    db.pickingEvento.count({ where }),
   594→    db.pickingEvento.findMany({ where, orderBy: [{ fecha: 'asc' }, { operario: 'asc' }, { horaMin: 'asc' }], select: { fecha: true, operario: true, horaMin: true, bultos: true, soporte: true, circuito: true } }),
   595→  ])
   596→
   597→  if (!count) {
   598→    return { registros: 0, vacio: true as const }
   599→  }
   600→
   601→  // gaps entre eventos consecutivos por operario y dia
   602→  const buckets = [
   603→    { label: '0-1 min', min: 0, max: 1 },
   604→    { label: '1-2 min', min: 1, max: 2 },
   605→    { label: '2-5 min', min: 2, max: 5 },
   606→    { label: '5-10 min', min: 5, max: 10 },
   607→    { label: '10-20 min', min: 10, max: 20 },
   608→    { label: '20-60 min', min: 20, max: 60 },
   609→    { label: '> 60 min', min: 60, max: Infinity },
   610→  ]
   611→  const gapCounts = new Array(buckets.length).fill(0)
   612→  const soporteGapCounts = new Array(buckets.length).fill(0)
   613→  let gapSum = 0, gapN = 0, soporteGapSum = 0, soporteGapN = 0, cambiosSoporte = 0
   614→  const opAgg = new Map<string, { nombre: string; eventos: number; gapSum: number; gapN: number; cambioSop: number; bultos: number }>()
   615→  const serDia = new Map<string, { eventos: number; gapMin: number }>()
   616→
   617→  let prev: { operario: string; key: string; horaMin: number; soporte: string | null } | null = null
   618→  for (const r of ops) {
   619→    const opKey = r.operario
   620→    let o = opAgg.get(opKey)
   621→    if (!o) { o = { nombre: opKey, eventos: 0, gapSum: 0, gapN: 0, cambioSop: 0, bultos: 0 }; opAgg.set(opKey, o) }
   622→    o.eventos += 1
   623→    o.bultos += r.bultos ?? 0
   624→    const k = r.fecha.toISOString().slice(0, 10)
   625→    let s = serDia.get(k)
   626→    if (!s) { s = { eventos: 0, gapMin: 0 }; serDia.set(k, s) }
   627→    s.eventos += 1
   628→
   629→    if (prev && prev.operario === opKey && prev.key === k && r.horaMin != null && prev.horaMin != null) {
   630→      const gap = Math.max(0, r.horaMin - prev.horaMin)
   631→      if (gap > 720) {
   632→        // salto de jornada (>12 h): no es tiempo muerto, se ignora para las metricas
   633→        prev = { operario: opKey, key: k, horaMin: r.horaMin ?? -1, soporte: r.soporte }
   634→        continue
   635→      }
   636→      const bi = buckets.findIndex((b) => gap >= b.min && gap < b.max)
   637→      if (bi >= 0) gapCounts[bi]++
   638→      gapSum += gap; gapN++
   639→      s.gapMin += gap
   640→      o.gapSum += gap; o.gapN++
   641→      const cambioSop = prev.soporte != null && r.soporte != null && prev.soporte !== r.soporte
   642→      if (cambioSop) {
   643→        cambiosSoporte++
   644→        if (bi >= 0) soporteGapCounts[bi]++
   645→        soporteGapSum += gap; soporteGapN++
   646→        o.cambioSop++
   647→      }
   648→    }
   649→    prev = { operario: opKey, key: k, horaMin: r.horaMin ?? -1, soporte: r.soporte }
   650→  }
   651→
   652→  const serie = [...serDia.entries()].sort().map(([fecha, v]) => ({ fecha, eventos: v.eventos, gapPromedio: v.eventos > 1 ? +(v.gapMin / (v.eventos - 1)).toFixed(2) : null }))
   653→
   654→  return {
   655→    registros: count,
   656→    vacio: false as const,
   657→    operarios: opAgg.size,
   658→    gapPromedio: gapN ? +(gapSum / gapN).toFixed(2) : null,
   659→    gapMedianaEst: gapN ? +(gapSum / gapN).toFixed(2) : null,
   660→    cambiosSoporte,
   661→    gapPromedioCambioSoporte: soporteGapN ? +(soporteGapSum / soporteGapN).toFixed(2) : null,
   662→    distribucionGaps: buckets.map((b, i) => ({ bucket: b.label, cantidad: gapCounts[i] })),
   663→    distribucionGapsSoporte: buckets.map((b, i) => ({ bucket: b.label, cantidad: soporteGapCounts[i] })),
   664→    operariosTop: [...opAgg.values()].filter((o) => o.gapN > 30).map((o) => ({ operario: o.nombre, eventos: o.eventos, bultos: o.bultos, gapPromedio: +(o.gapSum / o.gapN).toFixed(2), cambiosSoporte: o.cambioSop })).sort((a, b) => b.eventos - a.eventos).slice(0, 15),
   665→    serie,
   666→  }
   667→}

   719→      prodSuperNeto: x._sum.tiempoSuperNeto && x._sum.tiempoSuperNeto > 0 ? r2((x._sum.bultos ?? 0) / x._sum.tiempoSuperNeto) : null,
   720→    }))
   721→    .sort((a, b) => b.bultos - a.bultos)
   722→
   723→  // serie mensual
   724→  const porMesMap = new Map<string, { bultos: number; tiempoTotal: number; tiempoMuerto: number; tiempoSuperNeto: number }>()
   725→  for (const r of filas) {
   726→    const k = dia(r.fecha).slice(0, 7)
   727→    let m = porMesMap.get(k)
   728→    if (!m) { m = { bultos: 0, tiempoTotal: 0, tiempoMuerto: 0, tiempoSuperNeto: 0 }; porMesMap.set(k, m) }
   729→    m.bultos += r.bultos ?? 0
   730→    m.tiempoTotal += r.tiempoTotal ?? 0
   731→    m.tiempoMuerto += r.tiempoMuerto ?? 0
   732→    m.tiempoSuperNeto += r.tiempoSuperNeto ?? 0
   733→  }
   734→  const porMes = [...porMesMap.entries()].sort(([a], [b]) => a.localeCompare(b)).map(([mes, m]) => ({
   735→    mes,
   736→    bultos: Math.round(m.bultos),
   737→    pctMuerto: m.tiempoTotal ? r1((m.tiempoMuerto / m.tiempoTotal) * 100) : null,
   738→    prodSuperNeto: m.tiempoSuperNeto > 0 ? r2(m.bultos / m.tiempoSuperNeto) : null,
   739→  }))
   740→
   741→  const operariosTop = ops
   742→    .map((x) => ({
   743→      operario: x.operario,
   744→      nombre: x.nombre || x.operario,
   745→      bultos: Math.round(x._sum.bultos ?? 0),
   746→      soportes: Math.round(x._sum.soportes ?? 0),
   747→      horasTotal: r1(x._sum.tiempoTotal ?? 0),
   748→      pctMuerto: x._sum.tiempoTotal ? r1(((x._sum.tiempoMuerto ?? 0) / x._sum.tiempoTotal) * 100) : null,
   749→      prodSuperNeto: x._sum.tiempoSuperNeto && x._sum.tiempoSuperNeto > 0 ? r2((x._sum.bultos ?? 0) / x._sum.tiempoSuperNeto) : null,
   750→    }))
   751→    .sort((a, b) => b.bultos - a.bultos)
   752→    .slice(0, 15)
   753→
   754→  return {
   755→    registros: tot._count,
   756→    vacio: false as const,
   757→    desde: tot._min.fecha ? dia(tot._min.fecha) : undefined,
   758→    hasta: tot._max.fecha ? dia(tot._max.fecha) : undefined,
   759→    kpis,
   760→    porSector,
   761→    porMes,
   762→    operariosTop,
   763→  }
   764→}
